6// DualMonoid<M> wraps a monoidal value M and flips the argument order of
7// combine. The identity element is the same as M's identity.
8//
9// This is the standard algebraic dual: if (M, ·, e) is a monoid then
10// (M, ·ᵒᵖ, e) where a ·ᵒᵖ b = b · a is also a monoid.
11//
12// split-by-measure style operations require >= on the measure type;
13// DualMonoid<M> forwards >= from M so those operations work on reversed
14// structures.
15
16#include <smd/typeclass/monoid.hpp>
17
18#include <concepts>
19
20namespace smd::typeclass {
21
22template <typename M>
23struct DualMonoid {
24 M value;
25 friend auto operator==(const DualMonoid &, const DualMonoid &)
26 -> bool = default;
27
28 // Forward >= from M so split-by-measure compiles on reversed structures.
29 friend auto operator>=(const DualMonoid &lhs, const DualMonoid &rhs) -> bool
30 requires requires(const M &a, const M &b) {
31 { a >= b } -> std::convertible_to<bool>;
32 }
33 {
34 return lhs.value >= rhs.value;
35 }
36};
37
38template <typename M>
39struct Monoid<DualMonoid<M>> {
40 constexpr auto identity() const -> DualMonoid<M> {
41 return DualMonoid<M>{monoid_v<M>.identity()};
42 }
43 constexpr auto combine(const DualMonoid<M> &lhs,
44 const DualMonoid<M> &rhs) const -> DualMonoid<M> {
45 // Flip the argument order — that's the entire mechanism.
46 return DualMonoid<M>{monoid_v<M>.combine(rhs.value, lhs.value)};
47 }
48};
