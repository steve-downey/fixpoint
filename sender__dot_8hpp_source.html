6// Render a concrete sender expression as a Graphviz DOT graph, in two
7// stages that split the compile-time work from the runtime work:
8//
9// 1. reify_sender: a *structural* anamorphism from the heterogeneous
10// sender expression (every node its own static type) into a
11// first-class Fix<RoseF<sender_node_info, ·>> value. This cannot be
12// the library unfold_fix -- an unfold's seed type is fixed through
13// the recursion, and here every child seed is a different sender
14// type -- so the coalgebra runs as template recursion instead, one
15// instantiation per node type. The projection of one node is the
16// standard's exposition-only sender decomposition (tag, data,
17// children...), reached through beman::execution's implementation of
18// it, get_sender_data. Nothing is copied, connected, or started;
19// children are observed by reference.
20//
21// 2. to_dot: an ordinary catamorphism (fold_fix) over the reified tree.
22// Node numbering must be globally unique, so the algebra's carrier is
23// State-shaped -- a function from the next free id to the emitted
24// text -- and the fold composes those functions; running the composed
25// program from 0 assigns preorder ids and emits nodes and edges in
26// one pass. The same trick as fold.hpp's LeftFoldProgram: when a fold
27// needs threading, fold to a *program* and feed it the initial state.
28//
29// After reification every algorithm the library has for Fix works on the
30// sender tree: the Foldable/Traversable instances of RoseF give label
31// collection and effectful relabeling for free, and any other scheme
32// (para, histo, ...) applies unchanged.

53/** What the DOT rendering keeps from one sender node. */
54struct sender_node_info {
55 std::string tag_name; // "just", "then", "when_all", ...; non-value

75/** Display name for a sender tag: its leaf identifier, with the
76 * completion channel appended when the tag is pinned to a non-value
77 * channel. Several adaptors share one tag template parameterized by
78 * channel -- then_t<set_error_t> is upon_error -- and the leaf name alone
79 * would render them all as "then". Everything is computed in constexpr
80 * contexts; see short_type_name<T>() for why that is required, not style.
81 */
82template <typename Tag>
83auto tag_display_name() -> std::string {
84 constexpr std::string_view full = smd::fixpoint::type_name<Tag>();
85 constexpr std::string_view leaf = smd::fixpoint::short_type_name<Tag>();
86 constexpr bool on_error =
87 full.find("set_error_t") != std::string_view::npos;
88 constexpr bool on_stopped =
89 full.find("set_stopped_t") != std::string_view::npos;
90 std::string name(leaf);
91 if constexpr (on_error) {
92 name += "/error";
93 } else if constexpr (on_stopped) {
94 name += "/stopped";
95 }
96 return name;
97}

102/** Reify a sender expression into a first-class SenderTree value.
103 * Deliberately unconstrained: a non-decomposable type becomes a leaf
104 * labeled with its own type name, which is also what keeps this header
105 * away from the completion-signature machinery.
106 */
107template <typename Sender>
108auto reify_sender(const Sender &sndr) -> SenderTree {
109 auto &&decomposed = ::beman::execution::detail::get_sender_data(sndr);
110 using Decomposed = std::remove_cvref_t<decltype(decomposed)>;
111
112 if constexpr (requires { decomposed.tag; }) {
113 sender_node_info info{
114 detail::tag_display_name<typename Decomposed::tag_type>(),
115 std::string(smd::fixpoint::short_type_name<
116 typename Decomposed::data_type>())};
117 if (info.data_type == "make_sender_empty") {
118 info.data_type.clear(); // when_all's "no payload" marker
119 }
120 std::vector<SenderTree> children;
121 std::apply(
122 [&](const auto &...child) {
123 children.reserve(sizeof...(child));
124 (children.push_back(reify_sender(child)), ...);
125 },
126 decomposed.children);
127 return smd::fixpoint::wrap_fix<SenderTreeF>(
128 SenderTreeF<SenderTree>{std::move(info), std::move(children)});
129 } else {
130 return smd::fixpoint::wrap_fix<SenderTreeF>(SenderTreeF<SenderTree>{
131 sender_node_info{std::string(smd::fixpoint::short_type_name<
132 std::remove_cvref_t<Sender>>()),
133 std::string{}},
134 {}});
135 }
136}

141/** One emitted subtree: the id its root received, the next free id after
142 * the whole subtree, and the accumulated node and edge lines.
143 */
144struct dot_result {
145 int node_id;
146 int next_id;
147 std::string text;
148};
149
150/** The fold's State-shaped carrier: next free id -> emitted subtree. */
151struct dot_program {
152 std::function<dot_result(int)> run;
153};

155/** Escape a label for a double-quoted DOT string. */
156inline auto dot_escape(std::string_view text) -> std::string {
157 std::string out;
158 out.reserve(text.size());
159 for (char c : text) {
160 switch (c) {
161 case '"':
162 out += "\\\"";
163 break;
164 case '\\':
165 out += "\\\\";
166 break;
167 case '\n':
168 out += "\\n";
169 break;
170 default:
171 out += c;
172 }
173 }
174 return out;
175}

180/** Render a SenderTree as a Graphviz digraph. Ids are assigned in
181 * preorder -- the root is n0, each subtree owns a contiguous id range --
182 * by folding to a dot_program and running it from 0.
183 */
184inline auto to_dot(const SenderTree &tree) -> std::string {
185 auto algebra = [](const SenderTreeF<detail::dot_program> &layer)
186 -> detail::dot_program {
187 return detail::dot_program{[label = layer.label, kids = layer.children](
188 int next) -> detail::dot_result {
189 const int my_id = next;
190 int cursor = next + 1;
191 std::string text = " n" + std::to_string(my_id) + " [label=\"" +
192 detail::dot_escape(label.tag_name);
193 if (!label.data_type.empty()) {
194 text += "\\n" + detail::dot_escape(label.data_type);
195 }
196 text += "\"];\n";
197 for (const auto &kid : kids) {
198 detail::dot_result sub = kid.run(cursor);
199 cursor = sub.next_id;
200 text += sub.text;
201 text += " n" + std::to_string(my_id) + " -> n" +
202 std::to_string(sub.node_id) + ";\n";
203 }
204 return {my_id, cursor, std::move(text)};
205 }};
206 };
207 detail::dot_program program =
208 smd::fixpoint::fold_fix<detail::dot_program>(algebra, tree);
209 return "digraph G {\n node [shape=box];\n" + program.run(0).text + "}\n";
210}
