19/** Adapter: supplies `pure` (delegated to Impl) and synthesizes `apply` from
20 * `bind` + `pure` so a Monad can also serve as an Applicative base.
21 * ap mf mx = mf >>= \f -> mx >>= \a -> pure (f a)
22 *
23 * constexpr (design D10, S14): `apply` was not marked constexpr before S14
24 * (gana.t.cpp's ana_via_gana constexpr smoke test is the first caller to
25 * invoke `join` in a constant expression, via
26 * `monad_typeclass<MSeed>.join(...)` -- generalized.hpp) -- a pre-existing
27 * gap, not a new requirement; adding `constexpr` here is purely additive
28 * (Impl::bind/pure, which this forwards to, were already constexpr in every
29 * instance in this codebase).
30 */
31template <class Impl>
32struct monad_applicative_adapter : protected Impl {
33 using Impl::pure;
34
35 template <class MF, class MA>
36 constexpr auto apply(this auto &&self, MF &&mf, MA &&ma) {
37 return self.bind(std::forward<MF>(mf), [&self, &ma](auto &&f) {
38 return self.bind(ma, [&self, &f](auto &&a) {
39 return self.pure(std::invoke(std::forward<decltype(f)>(f),
40 std::forward<decltype(a)>(a)));
41 });
42 });
43 }
44};

46/** CRTP base for Monad instances.
47 * `Impl` must provide `pure(value)` and `bind(ma, f)`.
48 * `apply` is synthesized; `join` and `kleisli` are derived.
49 * Structurally chains onto `Applicative` (via `monad_applicative_adapter`),
50 * which itself chains onto `Functor`, so the full Functor + Applicative
51 * surface is also available on a Monad object.
52 */
53template <class Impl>
54struct Monad : protected Applicative<monad_applicative_adapter<Impl>> {
55 static_assert(!std::is_same_v<Impl, std::false_type>,
56 "No monad_typeclass<T> specialization found. "
57 "Specialize smd::typeclass::monad_typeclass<T> for your "
58 "type T and provide pure(...) and bind(...) operations.");
59
60 // Re-expose the full Functor + Applicative surface (protected
61 // inheritance hides it by default); Monad's own derived ops (join,
62 // kleisli, bind_with) stay declared below as normal public members.
63 using Applicative<monad_applicative_adapter<Impl>>::apply;
64 using Applicative<monad_applicative_adapter<Impl>>::apply_pure;
65 using Applicative<monad_applicative_adapter<Impl>>::apply_pure_with;
66 using Applicative<monad_applicative_adapter<Impl>>::ap;
67 using Applicative<monad_applicative_adapter<Impl>>::discard_first;
68 using Applicative<monad_applicative_adapter<Impl>>::discard_second;
69 using Applicative<monad_applicative_adapter<Impl>>::fmap;
70 using Applicative<monad_applicative_adapter<Impl>>::invoke;
71 using Applicative<monad_applicative_adapter<Impl>>::invoke_with;
72 using Applicative<monad_applicative_adapter<Impl>>::lift;

79 // join: flatten nested monad.
80 // join mma = mma >>= id
81 template <class MMA>
82 constexpr auto join(this auto &&self, MMA &&mma) {
83 return self.bind(std::forward<MMA>(mma),
84 [](auto &&inner) { return inner; });
85 }
86
87 // kleisli: forward Kleisli composition (>=>).
88 // (f >=> g) a = f a >>= g
89 template <class F, class G>
90 constexpr auto kleisli(this auto &&self, F f, G g) {
91 return [&self, f = std::move(f), g = std::move(g)](auto &&a) {
92 return self.bind(f(std::forward<decltype(a)>(a)), g);
93 };
94 }
95
96 // bind_with: explicit monad object override.
97 template <class MONAD_MAP, class MA, class F>
98 constexpr auto bind_with(this auto &&, const MONAD_MAP &monad_map, MA &&ma,
99 F &&f) {
100 return monad_map.bind(std::forward<MA>(ma), std::forward<F>(f));
101 }
102};

108// -- std::optional monad instance --
109// Delegates pure to the existing applicative_typeclass.
110
111template <class VALUE_TYPE>
112struct OptionalMonadImpl {
113 using element_type = VALUE_TYPE;
114
115 template <class VALUE>
116 auto pure(this auto &&, VALUE &&value)
117 -> std::optional<remove_cvref_t<VALUE>> {
118 return applicative_typeclass<std::optional<VALUE_TYPE>>.pure(
119 std::forward<VALUE>(value));
120 }
121
122 template <class A, class F>
123 auto bind(this auto &&, const std::optional<A> &ma, F &&f) {
124 using Result = remove_cvref_t<std::invoke_result_t<F, const A &>>;
125 if (!ma)
126 return Result{};
127 return Result{std::invoke(std::forward<F>(f), *ma)};
128 }
129};
130
131template <class VALUE_TYPE>
132struct OptionalMonadMap : Monad<OptionalMonadImpl<VALUE_TYPE>> {
133 using OptionalMonadImpl<VALUE_TYPE>::bind;
134 using OptionalMonadImpl<VALUE_TYPE>::pure;
135};
136
137/** Monad instance for `std::optional<VALUE_TYPE>`. */
138template <class VALUE_TYPE>
139inline constexpr auto monad_typeclass<std::optional<VALUE_TYPE>> =
140 OptionalMonadMap<VALUE_TYPE>{};

144/** Sequences a monadic value `ma` through function `f` (Haskell's `>>=`). */
145template <class MA, class F>
146constexpr auto mbind(MA &&ma, F &&f) {
147 const auto &map = monad_typeclass<remove_cvref_t<MA>>;
148 return map.bind(std::forward<MA>(ma), std::forward<F>(f));
149}
150
151/** Flattens a nested monadic value; equivalent to `bind(mma, id)`. */
152template <class MMA>
153constexpr auto join(MMA &&mma) {
154 const auto &map = monad_typeclass<remove_cvref_t<MMA>>;
155 return map.join(std::forward<MMA>(mma));
156}
