19/** mutu :: (f (a, b) -> a) -> (f (a, b) -> b) -> t -> (a, b)
20 *
21 * Equation: mutu f g = fold_fix(λx. pair(f(x), g(x)))
22 *
23 * @tparam A first fold's carrier (D5: explicit)
24 * @tparam B second fold's carrier (D5: explicit)
25 * @param alg_a F<std::pair<A, B>> -> A
26 * @param alg_b F<std::pair<A, B>> -> B
27 */
28template <class A, class B, template <class> class F, class AlgA, class AlgB>
29constexpr auto mutu(const AlgA &alg_a, const AlgB &alg_b, const Fix<F> &tree)
30 -> std::pair<A, B> {
31 using Carrier = std::pair<A, B>;
32 auto combined = [&](const F<Carrier> &layer) -> Carrier {
33 return Carrier{alg_a(layer), alg_b(layer)};
34 };
35 return fold_fix<Carrier>(combined, tree);
36}
