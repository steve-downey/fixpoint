6// FD10's motivating example, consolidated (S08). This is EXAMPLE / test-
7// support vocabulary, deliberately NOT a library header (it is not in the
8// `smd_fixpoint_headers` FILE_SET): the retry-with-backoff program, its two
9// domain signatures (Clock + Network), the domain data types, and the
10// Cofree-pairing mock (RetryScript coalgebra) that Run 1 / S06 drive.
11//
12// Kept beman-free on purpose: it is included by the beman-free
13// `smd_fixpoint_test` (freer_cosignature.t.cpp, the pairing interpreter) as
14// well as by the beman-linked S/R consumers. The sender/receiver handlers
15// (which DO need beman) live in the companion `freer_retry_handlers.hpp`.
16//
17// Everything here lives in the NAMED namespace `retry_example` (not an
18// anonymous one) so all translation units that include it share ONE set of
19// types and ONE program definition -- that is the whole point of S08 item 1
20// ("one canonical spelling"). Free functions are `inline` for the same
21// reason (ODR across the several TUs that include this).
22//
23// Canonical shape decision (S08): the recursive `retry_attempt` spelling
24// (S07's, cleaner than S06's fully-unrolled triple-nested mbind), and the
25// program FETCHES the time first (`send<Row>(Now{})`) before retrying --
26// FD10's narrative is "fetch time, retry with backoff", and keeping the
27// Now{} fetch means the program exercises BOTH Clock operations (Now and
28// SleepFor), not just SleepFor, so "Clock fully mocked" is a real claim
29// rather than half the signature going untouched. Both interpreters (Cofree
30// pairing and live S/R) therefore assert the SAME six-op trace:
31// Now(), Send(hello), SleepFor(1), Send(hello), SleepFor(2), Send(hello)
32

108/** One attempt: send the request; on success, the value is the reply's
109 * length; on failure, back off `attempt` seconds (SleepFor) and retry, up
110 * to three attempts total (1s after attempt 1, 2s after attempt 2). The
111 * value on total failure is -1; there is no fourth attempt. Composed with
112 * `mbind` + `pure` only (FD4): every continuation captures copyable state,
113 * so DEV-S05-1's move-only-continuation wall never bites here.
114 */
115inline auto retry_attempt(int attempt) -> RowFree {
116 return smd::typeclass::mbind(
117 smd::fixpoint::send<Row>(Send{request{"hello"}}),
118 [attempt](std::expected<reply, net_error> res) -> RowFree {
119 if (res) {
120 return smd::fixpoint::pure_free<Row::template type>(
121 static_cast<int>(res->body.size()));
122 }
123 if (attempt >= 3) {
124 return smd::fixpoint::pure_free<Row::template type>(
125 -1); // no fourth attempt
126 }
127 return smd::typeclass::mbind(
128 smd::fixpoint::send<Row>(SleepFor{attempt}),
129 [attempt](smd::fixpoint::unit) -> RowFree {
130 return retry_attempt(attempt + 1);
131 });
132 });
133}

135/** FD10: fetch the current time, then retry-with-backoff. The fetched time
136 * is a real (if here unused) start timestamp -- its point is that the Clock
137 * signature is exercised in full, and that both Clock operations flow
138 * through whichever handler is stacked underneath.
139 */
140inline auto retry_program() -> RowFree {
141 return smd::typeclass::mbind(
142 smd::fixpoint::send<Row>(Now{}),
143 [](time_point) -> RowFree { return retry_attempt(1); });
144}

146// -- The Cofree-pairing mock (Run 1 / S06): a scripted Network failing ---
147// -- twice then succeeding, paired with a virtual-time Clock, in ONE -----
148// -- Cofree value (FD8). ------------------------------------------------
149
150/** Script state: how many Sends have been answered (fail the first two,
151 * succeed on the third -- FD10's "failing twice then succeeding") and the
152 * virtual clock's current time (advanced only by SleepFor -- virtual time,
153 * no sleep_for, no threads).
154 */
155struct RetryScript {
156 int send_count = 0;
157 int virtual_time = 0;
158};
159
160inline auto retry_coalgebra(const RetryScript &state)
161 -> smd::fixpoint::cosignature<Row>::type<RetryScript> {
162 using smd::fixpoint::responder;
163 return smd::fixpoint::cosignature<Row>::type<RetryScript>{
164 responder<Now, RetryScript>{
165 [state](Now) -> std::pair<Now::response, RetryScript> {
166 return {time_point{state.virtual_time}, state};
167 }},
168 responder<SleepFor, RetryScript>{
169 [state](SleepFor op) -> std::pair<SleepFor::response, RetryScript> {
170 RetryScript next = state;
171 next.virtual_time += op.d;
172 return {smd::fixpoint::unit{}, next};
173 }},
174 responder<Send, RetryScript>{
175 [state](Send) -> std::pair<Send::response, RetryScript> {
176 RetryScript next = state;
177 ++next.send_count;
178 if (next.send_count <= 2) {
179 return {std::unexpected(net_error{"boom"}), next};
180 }
181 return {reply{"hello-reply"}, next}; // length 11
182 }}};
183}
184
185inline auto retry_head_fn(const RetryScript &state) -> RetryScript {
186 return state;
187}
