15// Monoid pattern invariants:
16// - Monoid<T> is the customization point; specialize identity/combine together.
17// - monoid_v<T> is the canonical lookup object used by generic algorithms.
18// - identity and combine must stay coherent for a single associative law
19// domain.
20// - Prefer adding new Monoid<T> specializations over ad hoc free functions.
21
22/** Customization point for the Monoid typeclass.
23 * Specialize this struct for type `VALUE_TYPE` and provide
24 * `identity()` and `combine(lhs, rhs)` to make that type usable
25 * wherever a Monoid is required (e.g., as the result type of fold_map).
26 */
27template <class VALUE_TYPE>
28struct Monoid;

30/** Canonical lookup object for Monoid<VALUE_TYPE>; used by generic algorithms.
31 */
32template <class VALUE_TYPE>
33inline constexpr Monoid<VALUE_TYPE> monoid_v = Monoid<VALUE_TYPE>{};

35/** Opaque count accumulator; the Monoid combines by addition. */
36struct Count {
37 std::size_t d_value;
38
39 friend constexpr bool operator==(const Count &lhs,

43/** Monoid<Count>: identity is zero, combine adds counts. */
44template <>
45struct Monoid<Count> {
46 constexpr auto identity() const -> Count { return Count{0}; }
47
48 constexpr auto combine(const Count &lhs, const Count &rhs) const -> Count {
49 return Count{lhs.d_value + rhs.d_value};
50 }
51};
