22/** apo :: (a -> f (Either t a)) -> a -> t
23 *
24 * Equation: apo ψ = fix ∘ fmapF (either id (apo ψ)) ∘ ψ
25 *
26 * Written with `match` (design §5.2), the worker never inspects whether
27 * Left holds `Fix<F>` or `const Fix<F>&` — so coalgebras may return
28 * `F<either<const Fix<F>&, Seed>>` to graft an existing subtree with no
29 * intermediate copy (the zero-copy graft, §7.2): the Left branch simply
30 * returns its referent, and returning a `const Fix<F>&` from a
31 * `Fix<F>`-returning function is an ordinary copy-construction, not a
32 * move-from-const&.
33 *
34 * @param coalgebra Seed -> F<either<Fix<F>, Seed>> (or the reference-Left
35 * variant above)
36 */
37template <template <class> class F, class Coalgebra, class Seed>
38constexpr auto apo(const Coalgebra &coalgebra, const Seed &seed) -> Fix<F> {
39 auto layer = coalgebra(seed);
40 auto expanded = layer_fmap(
41 [&](const auto &step) -> Fix<F> {
42 return smd::typeclass::match(
43 step, [](const auto &subtree) -> Fix<F> { return subtree; },
44 [&](const auto &next_seed) -> Fix<F> {
45 return apo<F>(coalgebra, next_seed);
46 });
47 },
48 layer);
49 return wrap_fix<F>(std::move(expanded));
50}
