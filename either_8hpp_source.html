17// either<L, R> — the project's short-circuit sum type (design D4). We use it
18// instead of std::expected because std::expected's sides are asymmetric
19// (`unexpected` is a distinct wrapper only on the error side, error-side
20// mapping is spelled `transform_error`) and `expected<T, T>` — which arises
21// naturally, e.g. apo's `either<Fix<F>, Seed>` when a coalgebra's finished
22// subtree and its seed share a type — forces tag-only construction and
23// invites ambiguity in generic code. either<L, R> instead gives *both* sides
24// a distinct wrapper (Left<L>/Right<R>), so either<T, T> always constructs
25// and compares unambiguously.
26//
27// Fixed convention, identical to the Haskell sources this library is
28// modeled on: Left = short-circuit/stop, Right = continue recursing.
29// Instances below are right-biased (functor maps Right, monad's pure is
30// make_right and bind propagates Left), matching Haskell's Either.
31//
32// either is the D11 dual of std::pair: pair is the product (both components
33// always present), either is the coproduct (exactly one alternative is
34// present). The library treats them symmetrically:
35// introduction pair{a, b} <-> make_left / make_right
36// elimination .first / .second <-> match(e, on_left, on_right)
37// pairing fanout(f, g) <-> fanin(f, g)
38// one-sided map map_first(f, p) <-> map_left(f, e)
39// functor maps .second <-> maps Right
40// Reference-assignment semantics differ between the two by construction, not
41// convention: std::pair<T&, U> assigns *through* the referent (both slots
42// always exist, so there is always something to assign into); either<L&, R>
43// *rebinds* (the active alternative can change under assignment, so the old
44// referent may no longer be the right target) — see the Left<L&>/Right<R&>
45// partial specializations below, modeled on the reference-side of
46// std::optional<T&> per P2988R12
47// (https://www.open-std.org/jtc1/sc22/wg21/docs/papers/2025/p2988r12.pdf).

51/** The stop/short-circuit side of `either<L, R>`. Primary template: an
52 * aggregate holding `L` by value.
53 */
54template <class L>
55struct Left {
56 L value;
57
58 /** Uniform accessor so generic code (including the `Left<L&>`
59 * specialization below) never has to know whether the referent is held
60 * by value or by pointer. Library and test code should go through this
61 * (or the free function `left(e)`), never `.value`, so the two
62 * specializations stay interchangeable.
63 */
64 constexpr auto referent() const -> const L & { return value; }
65
66 friend constexpr bool operator==(const Left &, const Left &) = default;
67};
68
69/** Reference-side specialization: `Left<L&>` holds a pointer to the
70 * referent (P2988 model), not a real reference member, so that assignment
71 * can *rebind* rather than being deleted (a struct with a true reference
72 * member has no assignment operator). Binds lvalues only: the `L&&`
73 * constructor is deleted so a temporary can never be bound and then
74 * dangle. Copy/assign rebind the pointer (shallow: the wrapper's own
75 * constness only ever protects the pointer, never the referent — a
76 * `const Left<L&>` still yields a mutable `L&` via `referent()`, exactly
77 * like a `const` pointer variable still permitting mutation through it).
78 */
79template <class L>
80struct Left<L &> {
81 L *ptr;
82
83 constexpr explicit Left(L &referent) : ptr(&referent) {}
84 constexpr Left(L &&) = delete; // no binding to temporaries
85
86 constexpr Left(const Left &) = default; // rebinds
87 constexpr auto operator=(const Left &) -> Left & = default; // rebinds
88
89 constexpr auto referent() const -> L & { return *ptr; }
90
91 friend constexpr bool operator==(const Left &lhs, const Left &rhs) {
92 return *lhs.ptr == *rhs.ptr;
93 }
94};

96/** The continue side of `either<L, R>`. Primary template: an aggregate
97 * holding `R` by value. Mirrors `Left<L>` exactly.
98 */
99template <class R>
100struct Right {
101 R value;
102
103 constexpr auto referent() const -> const R & { return value; }
104
105 friend constexpr bool operator==(const Right &, const Right &) = default;
106};

108/** Reference-side specialization for the continue side; mirrors
109 * `Left<L&>`.
110 */
111template <class R>
112struct Right<R &> {
113 R *ptr;
114
115 constexpr explicit Right(R &referent) : ptr(&referent) {}
116 constexpr Right(R &&) = delete; // no binding to temporaries
117
118 constexpr Right(const Right &) = default; // rebinds
119 constexpr auto operator=(const Right &) -> Right & = default; // rebinds
120
121 constexpr auto referent() const -> R & { return *ptr; }
122
123 friend constexpr bool operator==(const Right &lhs, const Right &rhs) {
124 return *lhs.ptr == *rhs.ptr;
125 }
126};

129/** Symmetric sum type: exactly one of `Left<L>` (stop) or `Right<R>`
130 * (continue). `Left`/`Right` are distinct wrapper types on *both* sides, so
131 * `either<T, T>` constructs and compares unambiguously — the failure mode
132 * that ruled out `std::expected<T, T>`.
133 */
134template <class L, class R>
135struct either {
136 using left_type = L;
137 using right_type = R;
138
139 std::variant<Left<L>, Right<R>> node;
140
141 friend constexpr bool operator==(const either &, const either &) = default;
142};
143
144/** Constructs the stop side. The result-side type `R` is named first (it
145 * cannot be deduced from the argument) and `L` is deduced from `v` — or,
146 * for the reference-side instantiation, supplied explicitly as `L&` so `v`
147 * is forced to be an lvalue-reference parameter (an rvalue argument fails
148 * to bind, closing the dangling-temporary hole at the call site, not just
149 * inside `Left<L&>`).
150 */
151template <class R, class L>
152constexpr auto make_left(L v) -> either<L, R> {
153 return either<L, R>{std::variant<Left<L>, Right<R>>{Left<L>{v}}};
154}
155
156/** Constructs the continue side; dual to `make_left`. */
157template <class L, class R>
158constexpr auto make_right(R v) -> either<L, R> {
159 return either<L, R>{std::variant<Left<L>, Right<R>>{Right<R>{v}}};
160}

163/** True when `e` holds the stop (Left) side. */
164template <class L, class R>
165constexpr auto is_left(const either<L, R> &e) -> bool {
166 return std::holds_alternative<Left<L>>(e.node);
167}
168
169/** Observes the stop side. Declared to return `const L&`: when `L` is
170 * itself a reference type (the P2988 reference-side instantiation), a
171 * `const` applied to a reference type is dropped by the language, so this
172 * collapses to plain `L&` — the same signature yields value semantics for
173 * plain `L` and shallow-const reference semantics for `L&`, with no
174 * special-casing needed at the call site. Generic code therefore cannot
175 * tell the two instantiations apart, per design D4/§5.2.
176 */
177template <class L, class R>
178constexpr auto left(const either<L, R> &e) -> const L & {
179 return std::get<Left<L>>(e.node).referent();
180}
181
182/** Observes the continue side; dual to `left`. */
183template <class L, class R>
184constexpr auto right(const either<L, R> &e) -> const R & {
185 return std::get<Right<R>>(e.node).referent();
186}

188/** Maps the stop side only, leaving Right untouched; the dual of
189 * `map_first` (pair.hpp). No left-functor typeclass is needed for this —
190 * it is the one free function callers reach for directly (dist_gapo uses
191 * it).
192 */
193template <class Fn, class L, class R>
194constexpr auto map_left(Fn &&fn, const either<L, R> &e) {
195 using B = remove_cvref_t<std::invoke_result_t<Fn, const L &>>;
196 return std::visit(
197 smd::fixpoint::overloaded{
198 [&fn](const Left<L> &l) -> either<B, R> {
199 return either<B, R>{std::variant<Left<B>, Right<R>>{
200 Left<B>{std::invoke(fn, l.referent())}}};
201 },
202 [](const Right<R> &r) -> either<B, R> {
203 return either<B, R>{std::variant<Left<B>, Right<R>>{r}};
204 },
205 },
206 e.node);
207}

209/** The coproduct eliminator `[f, g]` (the copairing, D11's dual of
210 * building a pair): applies `on_left` to the unwrapped stop referent or
211 * `on_right` to the unwrapped continue referent, whichever is present.
212 * Both branches must return the same type. Scheme workers (apo, elgot,
213 * gana) should prefer this over hand-rolled `is_left` branching.
214 */
215template <class L, class R, class OnLeft, class OnRight>
216constexpr auto match(const either<L, R> &e, OnLeft &&on_left,
217 OnRight &&on_right) {
218 return std::visit(smd::fixpoint::overloaded{
219 [&on_left](const Left<L> &l) {
220 return std::invoke(on_left, l.referent());
221 },
222 [&on_right](const Right<R> &r) {
223 return std::invoke(on_right, r.referent());
224 },
225 },
226 e.node);
227}
228
229/** Returns the matcher `[f, g]` as a callable value; the dual of
230 * `fanout(f, g)` (pair.hpp).
231 */
232template <class F, class G>
233constexpr auto fanin(F f, G g) {
234 return [f = std::move(f), g = std::move(g)](const auto &e) {
235 return match(e, f, g);
236 };
237}

241template <class L, class R>
242struct EitherFunctorImpl {
243 template <class Fn>
244 constexpr auto fmap(this auto &&, Fn &&fn, const either<L, R> &e) {
245 using B = remove_cvref_t<std::invoke_result_t<Fn, const R &>>;
246 return std::visit(
247 smd::fixpoint::overloaded{
248 [](const Left<L> &l) -> either<L, B> {
249 return either<L, B>{std::variant<Left<L>, Right<B>>{l}};
250 },
251 [&fn](const Right<R> &r) -> either<L, B> {
252 return either<L, B>{std::variant<Left<L>, Right<B>>{
253 Right<B>{std::invoke(fn, r.referent())}}};
254 },
255 },
256 e.node);
257 }
258};

261struct EitherFunctorMap : Functor<EitherFunctorImpl<L, R>> {
262 using EitherFunctorImpl<L, R>::fmap;

270// -- Monad (right-biased: pure = make_right, bind propagates Left) --
271
272template <class L, class R>
273struct EitherMonadImpl {
274 template <class VALUE>
275 constexpr auto pure(this auto &&, VALUE &&value) {
276 return make_right<L>(std::forward<VALUE>(value));
277 }
278
279 // Generic over the incoming Right type (its own template parameter
280 // `R2`), with `L` fixed to the instance's own `L` — matching
281 // OptionalMonadImpl's `bind` being generic in its own template
282 // parameter rather than the instance-keying `VALUE_TYPE`. This is what
283 // lets the Monad CRTP's derived `join`/`apply`/`kleisli` invoke `bind`
284 // on the doubled structure `either<L, either<L, R2>>` they produce;
285 // `R` above only serves to key the `monad_typeclass<either<L, R>>`
286 // lookup below.
287 template <class R2, class F>
288 constexpr auto bind(this auto &&, const either<L, R2> &ma, F &&f) {
289 using Result = remove_cvref_t<std::invoke_result_t<F, const R2 &>>;
290 using ResultRight = typename Result::right_type;
291 return std::visit(
292 smd::fixpoint::overloaded{
293 [](const Left<L> &l) -> Result {
294 return Result{std::variant<Left<L>, Right<ResultRight>>{l}};
295 },
296 [&f](const Right<R2> &r) -> Result {
297 return std::invoke(f, r.referent());
298 },
299 },
300 ma.node);
301 }
302};
303
304template <class L, class R>
305struct EitherMonadMap : Monad<EitherMonadImpl<L, R>> {
306 using EitherMonadImpl<L, R>::bind;
307 using EitherMonadImpl<L, R>::pure;
308};
309
310/** Monad instance for `either<L, R>`: `pure` = `make_right`, `bind`
311 * propagates Left untouched — matching Haskell's Either.
312 */
313template <class L, class R>
314inline constexpr auto monad_typeclass<either<L, R>> = EitherMonadMap<L, R>{};
