6// Foldable and Traversable instances for std::vector.
7//
8// std::vector is the shared traversable *structure* for all three of Paper A's
9// motivating domains. Pairing this one Traversable instance with three
10// different applicative contexts gives the three front-door results through a
11// single API:
12//
13// traverse(f, vector<optional<T>>) -> optional<vector<T>> (fallible value)
14// traverse(f, vector<sender<T>>) -> sender<vector<T>> (deferred result)
15// traverse(f, vector<zip_list<T>>) -> zip_list<vector<T>> (lanewise / SIMD)
16//
17// The traverse primitive sequences each element's effect left-to-right, lifting
18// element results back into a vector while preserving the original shape.

33/** Foldable instance for std::vector: fold_map accumulates left-to-right. */
34template <class VALUE_TYPE>
35struct VectorFoldableImpl {
36 template <class FUNCTION>
37 auto fold_map(this auto &&, FUNCTION &&function,
38 const std::vector<VALUE_TYPE> &values) {
39 using Result =
40 remove_cvref_t<std::invoke_result_t<FUNCTION, const VALUE_TYPE &>>;
41 auto accumulated = monoid_identity<Result>();
42 for (const auto &value : values) {
43 accumulated = monoid_combine(std::move(accumulated),
44 std::invoke(function, value));
45 }
46 return accumulated;
47 }
48};
49
50template <class VALUE_TYPE>
51struct VectorFoldableMap : Foldable<VectorFoldableImpl<VALUE_TYPE>> {
52 using VectorFoldableImpl<VALUE_TYPE>::fold_map;
53};

55/** Foldable instance for `std::vector<VALUE_TYPE>`. */
56template <class VALUE_TYPE>
57inline constexpr auto foldable_typeclass<std::vector<VALUE_TYPE>> =
58 VectorFoldableMap<VALUE_TYPE>{};

60/** Traversable instance for std::vector.
61 * The primitive `traverse` sequences each element's effect via the supplied
62 * applicative, collecting the in-context element results into a vector while
63 * preserving the original element order and count.
64 */
65template <class VALUE_TYPE>
66struct VectorTraversableImpl {
67 using element_type = VALUE_TYPE;
68
69 template <class APPLICATIVE, class FUNCTION>
70 auto traverse(this auto &&, const APPLICATIVE &applicative,
71 FUNCTION &&function, const std::vector<VALUE_TYPE> &values) {
72 using Effect =
73 remove_cvref_t<std::invoke_result_t<FUNCTION, const VALUE_TYPE &>>;
74 using Element = applicative_value_t<Effect>;
75
76 auto accumulated = applicative.pure(std::vector<Element>{});
77 for (const auto &value : values) {
78 accumulated = applicative.invoke(
79 [](std::vector<Element> collected, const Element &element) {
80 collected.push_back(element);
81 return collected;
82 },
83 accumulated, std::invoke(function, value));
84 }
85 return accumulated;
86 }
87};
88
89template <class VALUE_TYPE>
90struct VectorTraversableMap : Traversable<VectorTraversableImpl<VALUE_TYPE>> {
91 using VectorTraversableImpl<VALUE_TYPE>::traverse;
92};

94/** Traversable instance for `std::vector<VALUE_TYPE>`. */
95template <class VALUE_TYPE>
96inline constexpr auto traversable_typeclass<std::vector<VALUE_TYPE>> =
97 VectorTraversableMap<VALUE_TYPE>{};
