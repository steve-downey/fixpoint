18// Identity function for fold composition - no type erasure needed
19template <class STATE>
20struct IdentityFoldFunc {
21 constexpr auto operator()(STATE s) const -> STATE { return s; }
22};
23
24// Compose two fold functions without type erasure
25template <class F1, class F2>
26struct ComposedFoldFunc {
27 F1 d_f1;
28 F2 d_f2;
29
30 template <class STATE>
31 auto operator()(STATE s) const -> STATE {
32 return d_f2(d_f1(std::move(s)));
33 }
34};
35
36// Left fold composition (same order as f1 then f2)
37template <class STATE>
38struct LeftFoldProgram {
39 std::function<STATE(STATE)> d_run;
40
41 auto operator()(STATE state) const -> STATE {
42 return d_run(std::move(state));
43 }
44};

192// Foldable pattern invariants:
193// - Generic entry point is fold_map(F, T) via foldable_typeclass<T>.
194// - Instances provide an object with fold_map(F, T) and specialize
195// foldable_typeclass<Concrete>.
196// - Derived APIs (length, fold_left, fold_right, combine_all, any_of, all_of,
197// empty, to_vector, find_first) live on the same looked-up object.
198// - Traversal order is instance-defined but must be coherent per instance.
199
200/** CRTP base for Foldable instances.
201 * `Impl` must provide either `fold_map(f, container)` or `fold_right` +
202 * `element_type`; all other operations are derived from whichever is the
203 * primitive.
204 */
205template <class Impl>
206struct Foldable : protected Impl {
207 static_assert(
208 !std::is_same_v<Impl, std::false_type>,
209 "No foldable_typeclass<T> specialization found. "
210 "Specialize smd::typeclass::foldable_typeclass<T> for your "
211 "type T and provide fold_map(F, T) or fold_right(T, STATE, F) "
212 "+ element_type.");
213 // Alternate-core: Impl provides either fold_map or fold_right as primitive.
214 // The Map class's using-declaration selects which; the base derives the
215 // other. Haskell equivalent: {-# MINIMAL foldMap | foldr #-}
216
217 // Derived fold_map from fold_right. Active when a fold_right-primitive
218 // Impl's using-declaration shadows the base's derived fold_right with the
219 // real one. Requires element_type to deduce the monoid result type. foldMap
220 // f = foldr (\x acc -> f x <> acc) mempty
221 template <class F, class T>
222 auto fold_map(this auto &&self, F &&function, T &&value)
223 requires requires { typename Impl::element_type; }
224 {
225 using Result = remove_cvref_t<
226 std::invoke_result_t<F, const typename Impl::element_type &>>;
227 return self.fold_right(
228 std::forward<T>(value), monoid_identity<Result>(),
229 [&function](const auto &elem, Result acc) {
230 return monoid_combine(std::invoke(function, elem),
231 std::move(acc));
232 });
233 }

235 /** Returns the number of elements in the foldable container. */
236 template <class T>
237 auto length(this auto &&self, T &&value) -> std::size_t {
238 const auto count = self.fold_map([](const auto &) { return Count{1}; },
239 std::forward<T>(value));
240 return count.d_value;
241 }

243 /** Left-associative fold: applies `function(state, element)` for each
244 * element in traversal order, starting from `initial_state`.
245 */
246 template <class T, class STATE, class F>
247 auto fold_left(this auto &&self, T &&value, STATE initial_state,
248 F &&function) {
249 using StateType = remove_cvref_t<STATE>;
250 auto step = std::forward<F>(function);
251
252 const auto program = self.fold_map(
253 [&step](const auto &x) {
254 using ValueType = remove_cvref_t<decltype(x)>;
255 return detail::LeftFoldProgram<StateType>{
256 [x_copy = ValueType(x), &step](StateType s) {
257 return std::invoke(step, std::move(s), x_copy);
258 }};
259 },
260 std::forward<T>(value));
261
262 return program(StateType(std::move(initial_state)));
263 }
264
265 /** Right-associative fold: applies `function(element, state)` for each
266 * element in reverse traversal order, starting from `initial_state`.
267 */
268 template <class T, class STATE, class F>
269 auto fold_right(this auto &&self, T &&value, STATE initial_state,
270 F &&function) {
271 using StateType = remove_cvref_t<STATE>;
272 auto step = std::forward<F>(function);
273
274 const auto program = self.fold_map(
275 [&step](const auto &x) {
276 using ValueType = remove_cvref_t<decltype(x)>;
277 return detail::RightFoldProgram<StateType>{
278 [x_copy = ValueType(x), &step](StateType s) {
279 return std::invoke(step, x_copy, std::move(s));
280 }};
281 },
282 std::forward<T>(value));
283
284 return program(StateType(std::move(initial_state)));
285 }

326 /** Returns `true` if the container holds no elements. */

333 /** Collects all elements into a `std::vector` in traversal order. */
334 template <class T>
335 auto to_vector(this auto &&self, T &&value) {
336 return self.fold_map(
337 [](const auto &x) {
338 using ValueType = remove_cvref_t<decltype(x)>;
339 return std::vector<ValueType>{x};

341 std::forward<T>(value));

344 /** Returns the first element satisfying `predicate`, or an empty optional.
345 */
346 template <class T, class PREDICATE>
347 auto find_first(this auto &&self, T &&value, PREDICATE &&predicate) {
348 const auto result = self.fold_map(
349 [&predicate](const auto &x) {
350 using X = remove_cvref_t<decltype(x)>;
351 if (std::invoke(predicate, x)) {
352 return detail::First<X>{{x}};
353 }
354 return detail::First<X>{{}};
355 },
356 std::forward<T>(value));
357
358 return result.d_value;
359 }
