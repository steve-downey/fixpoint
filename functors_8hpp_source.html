35// 534bcaa9-4a28-453d-83c3-37d4f479bc8c
36struct Zero {
37 // Hand-written (not = default): Clang 22 eagerly instantiates a
38 // defaulted comparison operator's deleted-ness check inside a class
39 // template body, which forces completeness of any self-referential
40 // functor family wrapping this type (e.g. Cofree<NatF, Cofree<NatF,A>>,
41 // "squared" comonad-duplicate results, S13/S15) before that family has
42 // finished being defined — GCC defers this check, so it never hits the
43 // cycle. A plain friend body is only instantiated when actually
44 // ODR-used (both compilers agree on this), which breaks the cycle.
45 friend constexpr auto operator==(const Zero &, const Zero &) -> bool {
46 return true;
47 }
48};
49
50template <typename A>
51struct Succ {
52 smd::fixpoint::Box<A> pred;
53
54 // Hand-written for the same reason as Zero's above. Member-wise
55 // comparison of Box<A>: Box's own operator== already null-checks
56 // (box.hpp), so comparing the Box objects directly (not dereferencing)
57 // is exactly what `= default` would have produced.
58 friend constexpr auto operator==(const Succ &lhs, const Succ &rhs) -> bool {
59 return lhs.pred == rhs.pred;
60 }
61};
62
63template <typename A>
64using NatF = std::variant<Zero, Succ<A>>;
65
66using Nat = smd::fixpoint::Fix<NatF>;
67// 534bcaa9-4a28-453d-83c3-37d4f479bc8c end

155 friend constexpr auto operator==(const Nil &, const Nil &) -> bool {

160template <typename E, typename A>
161struct Cons {
162 E head;
163 smd::fixpoint::Box<A> tail;
164
165 // Hand-written for the same reason as Nil's above. Member-wise, in
166 // declaration order, short-circuiting on the first false — exactly
167 // what `= default` would produce.
168 friend constexpr auto operator==(const Cons &lhs, const Cons &rhs) -> bool {
169 return lhs.head == rhs.head && lhs.tail == rhs.tail;
170 }
171};

222/** Unfold: build an IntList from a std::vector<int>, front to back (ana). */
223constexpr auto list_from_vector(const std::vector<int> &v) -> IntList {
224 return smd::fixpoint::unfold_fix<IntListF>(
225 [&v](std::size_t i) -> IntListF<std::size_t> {
226 if (i >= v.size()) {
227 return Nil<int>{};
228 }
229 return Cons<int, std::size_t>{
230 v[i], smd::fixpoint::make_box<std::size_t>(i + 1)};
231 },
232 std::size_t{0});
233}
234
235/** Fold: collect an IntList into a std::vector<int> (cata). */
236constexpr auto list_to_vector(const IntList &list) -> std::vector<int> {
237 return smd::fixpoint::fold_fix<std::vector<int>>(
238 [](const IntListF<std::vector<int>> &layer) -> std::vector<int> {
239 return std::visit(
240 smd::fixpoint::overloaded{
241 [](const Nil<int> &) -> std::vector<int> { return {}; },
242 [](const Cons<int, std::vector<int>> &c)
243 -> std::vector<int> {
244 std::vector<int> result{c.head};
245 result.insert(result.end(), c.tail->begin(),
246 c.tail->end());
247 return result;
248 },
249 },
250 layer);
251 },
252 list);
253}
