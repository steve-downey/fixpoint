34// 534bcaa9-4a28-453d-83c3-37d4f479bc8c
35struct Zero {
36 // Hand-written (not = default): Clang 22 eagerly instantiates a
37 // defaulted comparison operator's deleted-ness check inside a class
38 // template body, which forces completeness of any self-referential
39 // functor family wrapping this type (e.g. Cofree<NatF, Cofree<NatF,A>>,
40 // "squared" comonad-duplicate results, S13/S15) before that family has
41 // finished being defined — GCC defers this check, so it never hits the
42 // cycle. A plain friend body is only instantiated when actually
43 // ODR-used (both compilers agree on this), which breaks the cycle.
44 friend constexpr auto operator==(const Zero &, const Zero &) -> bool {
45 return true;
46 }
47};
48
49template <typename A>
50struct Succ {
51 smd::fixpoint::Box<A> pred;
52
53 // Hand-written for the same reason as Zero's above. Member-wise
54 // comparison of Box<A>: Box's own operator== already null-checks
55 // (box.hpp), so comparing the Box objects directly (not dereferencing)
56 // is exactly what `= default` would have produced.
57 friend constexpr auto operator==(const Succ &lhs, const Succ &rhs) -> bool {
58 return lhs.pred == rhs.pred;
59 }
60};
61
62template <typename A>
63using NatF = std::variant<Zero, Succ<A>>;

65using Nat = smd::fixpoint::Fix<NatF>;

119/** Unfold: build a Nat counting down from @p n (ana). */
120constexpr auto nat_from_int(int n) -> Nat {
121 return smd::fixpoint::unfold_fix<NatF>(

127 },

147// ListF<E, ·> — element-parameterized cons-lists (design D3).

150// 9647fdcb-1a8f-4b77-82e6-97b5f8632788
151template <typename E>
152struct Nil {
153 // Hand-written for the same reason as Zero's above (S07).
154 friend constexpr auto operator==(const Nil &, const Nil &) -> bool {
155 return true;
156 }
157};
158
159template <typename E, typename A>
160struct Cons {
161 E head;
162 smd::fixpoint::Box<A> tail;
163
164 // Hand-written for the same reason as Nil's above. Member-wise, in
165 // declaration order, short-circuiting on the first false — exactly
166 // what `= default` would produce.
167 friend constexpr auto operator==(const Cons &lhs, const Cons &rhs) -> bool {
168 return lhs.head == rhs.head && lhs.tail == rhs.tail;
169 }
170};
171
172template <typename E, typename A>
173using ListF = std::variant<Nil<E>, Cons<E, A>>;
174
175template <typename A>
176using IntListF = ListF<int, A>;

178using IntList = smd::fixpoint::Fix<IntListF>;

221/** Unfold: build an IntList from a std::vector<int>, front to back (ana). */
222constexpr auto list_from_vector(const std::vector<int> &v) -> IntList {
223 return smd::fixpoint::unfold_fix<IntListF>(
224 [&v](std::size_t i) -> IntListF<std::size_t> {
225 if (i >= v.size()) {
226 return Nil<int>{};
227 }
228 return Cons<int, std::size_t>{
229 v[i], smd::fixpoint::make_box<std::size_t>(i + 1)};
230 },
231 std::size_t{0});
232}
233
234/** Fold: collect an IntList into a std::vector<int> (cata). */
235constexpr auto list_to_vector(const IntList &list) -> std::vector<int> {
236 return smd::fixpoint::fold_fix<std::vector<int>>(
237 [](const IntListF<std::vector<int>> &layer) -> std::vector<int> {
238 return std::visit(
239 smd::fixpoint::overloaded{
240 [](const Nil<int> &) -> std::vector<int> { return {}; },
241 [](const Cons<int, std::vector<int>> &c)
242 -> std::vector<int> {
243 std::vector<int> result{c.head};
244 result.insert(result.end(), c.tail->begin(),
245 c.tail->end());
246 return result;
247 },
248 },
249 layer);
250 },
251 list);
252}
