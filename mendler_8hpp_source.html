6// Mendler-style catamorphism and histomorphism (design §7.7). Every other
7// scheme in this module recurses by handing a plain algebra to fold_fix/
8// unfold_fix/refold, which fmap the recursive call over a pre-computed layer
9// via smd::typeclass::functor_typeclass (design D2/§4). mcata and mhisto work
10// the other way around: the algebra itself receives the recursive call (and,
11// for mhisto, the unroller) as an explicit callable argument, and calls it on
12// whichever children it likes. Consequently **no functor_typeclass instance
13// is ever looked up for F** — mcata/mhisto compile and run over any base
14// functor F, instanced or not.
15//
16// Equations:
17// mcata Φ t = Φ (mcata Φ) (unfix t)
18// mhisto Φ t = Φ (mhisto Φ) unfix (unfix t)
19// mhisto's extra `unfix` argument lets the algebra peel further layers off
20// any child on demand — course-of-values history without ever building a
21// Cofree annotation (contrast with histo.hpp, S07).
22//
23// The literature type is rank-2 polymorphic in the recursive-call argument
24// (`forall y. (y -> c) -> f y -> c`): well-typed algebras may only ever
25// apply that argument to a child, never otherwise inspect it. C++ has no
26// rank-2 polymorphism and cannot enforce this: a mischievous algebra could
27// still reach into a child Fix<F> directly (e.g. via unwrap_fix) and bypass
28// the callable entirely. Nothing here stops that — the discipline "the
29// recurse/unroll callable is the *only* thing you may do with a child" is by
30// convention only. Demonstrating that discipline (and that it costs nothing
31// in expressiveness) is the pedagogical point of mendler_eval.cpp.

38/** mcata :: (forall y. (y -> c) -> f y -> c) -> Fix f -> c
39 * @tparam Result the fold's result type (design D5: explicit)
40 * @param phi called as phi(recurse, layer): `recurse` is a callable
41 * const Fix<F>& -> Result (mcata Φ, partially applied);
42 * `layer` is const F<Fix<F>>&, the tree's outermost layer.
43 * phi decides which children (if any) to fold via `recurse` —
44 * no functor_typeclass<F<...>> instance is consulted here.
45 */
46template <class Result, template <class> class F, class MAlgebra>
47constexpr auto mcata(const MAlgebra &phi, const Fix<F> &tree) -> Result {
48 auto recurse = [&phi](const Fix<F> &child) -> Result {
49 return mcata<Result, F>(phi, child);
50 };
51 return phi(recurse, unwrap_fix(tree));
52}

56/** mhisto :: (forall y. (y -> c) -> (y -> f y) -> f y -> c) -> Fix f -> c
57 * @tparam Result the fold's result type (design D5: explicit)
58 * @param phi called as phi(recurse, unroll, layer): `recurse` is
59 * const Fix<F>& -> Result (mhisto Φ, partially applied);
60 * `unroll` is const Fix<F>& -> const F<Fix<F>>& (i.e.
61 * unwrap_fix) — it returns a reference *into* the tree, which
62 * is fine here since the tree the fold started from outlives
63 * the whole call; `layer` is const F<Fix<F>>&. phi may call
64 * `unroll` on any child it holds to look further down without
65 * committing to folding it via `recurse` — no
66 * functor_typeclass<F<...>> instance is consulted here either.
67 */
68template <class Result, template <class> class F, class MAlgebra>
69constexpr auto mhisto(const MAlgebra &phi, const Fix<F> &tree) -> Result {
70 auto recurse = [&phi](const Fix<F> &child) -> Result {
71 return mhisto<Result, F>(phi, child);
72 };
73 auto unroll = [](const Fix<F> &child) -> const F<Fix<F>> & {
74 return unwrap_fix(child);
75 };
76 return phi(recurse, unroll, unwrap_fix(tree));
77}
