13// layer_fmap is the single point where scheme bodies bridge into
14// smd::typeclass functor dispatch: instead of threading an explicit fmap_fn
15// through every scheme, they call layer_fmap(fn, layer) and the typeclass
16// lookup keyed on std::remove_cvref_t<Layer> finds the right fmap.
17//
18// It offers the three lookup modes of the typeclass-object design
19// ("Writing Algorithms with Typeclass Objects"):
20//
21// 1. Implicit lookup layer_fmap(fn, layer)
22// The Typeclass NTTP defaults to functor_typeclass<Layer>; both call
23// arguments are deduced. This is what every scheme uses.
24//
25// 2. NTTP pinning layer_fmap<Layer, my_functor>(fn, layer)
26// Bind a specific instance as the non-type template parameter. Only
27// Layer (a spellable type) and the object are named; Fn still deduces,
28// so the callable stays an inline lambda. Layer precedes the NTTP for
29// exactly this reason -- keeping the callable a trailing *deduced*
30// parameter is what lets an unspellable lambda type through.
31//
32// 3. Explicit object layer_fmap(my_functor, fn, layer)
33// Pass the instance as an ordinary first argument. The instance need
34// not be registered in functor_typeclass at all -- the escape hatch for
35// code (tests above all) that wants the machinery without owning, or
36// risking an ODR clash on, the global specialization.
37
38// d1f5f4d6-a17a-47c6-b5cb-b97d08a0af6b
39/** Concept satisfied by a functor typeclass *object* — anything exposing an
40 * `fmap(fn, layer)` member. Disambiguates the explicit-object overload
41 * (mode 3, instance first) from the implicit/pinned lookup form. */
42template <class Typeclass, class Fn, class Layer>
43concept functor_instance_for =
44 requires(const Typeclass &tc, Fn &&fn, const Layer &layer) {
45 tc.fmap(std::forward<Fn>(fn), layer);
46 };
47
48/** Apply the functor instance for @p layer's concrete type to @p fn.
49 *
50 * Modes 1 (implicit lookup) and 2 (NTTP pinning): the @p Typeclass reference
51 * NTTP defaults to `functor_typeclass<remove_cvref_t<Layer>>` but may be
52 * pinned at the call site as `layer_fmap<Layer, instance>(fn, layer)`.
53 *
54 * The NTTP is `const auto&` — a *reference* NTTP binding the `inline
55 * constexpr` typeclass variable's static storage. A by-value `auto` NTTP
56 * would instead require the instance to be a *structural* type, which the
57 * `Functor`/`Monad` CRTP instances are not (their `protected` base is
58 * non-structural); the reference form sidesteps that entirely. @p Layer
59 * precedes the NTTP so @p Fn can trail as a deduced parameter and an inline
60 * lambda never has to be spelled to reach the pin.
61 */
62template <class Layer,
63 const auto &Typeclass =
64 smd::typeclass::functor_typeclass<std::remove_cvref_t<Layer>>,
65 class Fn>
66constexpr auto layer_fmap(Fn &&fn, const Layer &layer) {
67 return Typeclass.fmap(std::forward<Fn>(fn), layer);
68}

71// FD4: consuming (rvalue) traversal. A move-only, lazily-mapped layer
72// (S03's Coyoneda impure_node) cannot compile against the const-lvalue
73// overloads above -- they rebuild by copy. These overloads take the layer
74// by rvalue and forward std::move(layer) into Typeclass.fmap, so `bind`
75// (free.hpp) and later the Coyoneda instance can move through instead of
76// copying. `Layer` is a genuine forwarding reference here, constrained to
77// `!std::is_lvalue_reference_v<Layer>` so it never wins over the const&
78// overloads above for an lvalue argument -- ordinary overload resolution
79// then prefers this rvalue-parameter overload over the const&-binds-to-
80// rvalue fallback for an actual rvalue call. A pure-data instance that
81// only ever wrote a `const Layer&` fmap is still reachable through this
82// overload: `Typeclass.fmap(fn, std::move(layer))` binds the moved-from
83// rvalue to that `const Layer&` parameter same as any rvalue would.
84template <class Layer,
85 const auto &Typeclass =
86 smd::typeclass::functor_typeclass<std::remove_cvref_t<Layer>>,
87 class Fn>
88 requires(!std::is_lvalue_reference_v<Layer>)
89constexpr auto layer_fmap(Fn &&fn, Layer &&layer) {
90 return Typeclass.fmap(std::forward<Fn>(fn), std::forward<Layer>(layer));
91}

94/** Mode 3 (explicit object): apply the functor object @p tc directly instead
95 * of consulting the global registry. The typeclass object is the first
96 * argument — it threads ahead of the data it maps over, so a scheme can
97 * carry an unregistered instance down through its recursion. Constrained on
98 * `functor_instance_for` so it never competes with the two-argument
99 * implicit/pinned form above.
100 */
101template <class Typeclass, class Fn, class Layer>
102 requires functor_instance_for<Typeclass, Fn, Layer>
103constexpr auto layer_fmap(const Typeclass &tc, Fn &&fn, const Layer &layer) {
104 return tc.fmap(std::forward<Fn>(fn), layer);
105}

108// FD4: rvalue mirror of `functor_instance_for` for mode 3's consuming
109// overload below -- checks that @p tc.fmap accepts @p layer by rvalue
110// instead of `const Layer&`. Constrained to `!std::is_lvalue_reference_v`
111// for the same reason as the mode 1/2 rvalue overload above: keep the
112// existing const-lvalue mode-3 overload winning for lvalue calls.
113template <class Typeclass, class Fn, class Layer>
114concept functor_instance_for_rvalue =
115 (!std::is_lvalue_reference_v<Layer>) &&
116 requires(const Typeclass &tc, Fn &&fn, Layer &&layer) {
117 tc.fmap(std::forward<Fn>(fn), std::forward<Layer>(layer));
118 };
