30/** run(handler, prog) -> A: interpret a Freer<Sig, A> program to completion
31 * against `handler`.
32 *
33 * `handler` is any callable set covering every operation in `Sig` -- an
34 * `overloaded` of one lambda per Op is the intended shape (FD1: handlers
35 * pattern-match on requests, so `std::visit` over the node variant is the
36 * dispatch). A missing case is a compile error at this call: the
37 * static_assert below fires naming the uncovered `impure_node<Op, X>` in
38 * the instantiation backtrace, rather than a generic "no matching
39 * `std::visit` alternative" failure.
40 *
41 * A LOOP, not recursion (FD6, pitfall list): each iteration handles
42 * exactly one suspended operation and rebinds `current` to the resumed
43 * result, so the C++ call stack never grows with the number of effects a
44 * program performs -- a hand-recursive `run` would stack one frame per
45 * effect and Asan would not tolerate a program of any real depth.
46 * `X = Freer<Sig, A>` is fixed for the whole loop: `Free` is
47 * self-referential through `F<Free<F,A>>`, so every layer's recursive
48 * position already is `Freer<Sig, A>` itself, and resuming a continuation
49 * always hands back exactly that type -- no re-templating between
50 * iterations.
51 */
52template <class Sig, class A, class Handler>
53auto run(Handler &&handler, Freer<Sig, A> &&prog) -> A {
54 using X = Freer<Sig, A>;
55 using Layer = typename Sig::template type<X>;
56
57 Freer<Sig, A> current = std::move(prog);
58 while (!is_pure(current)) {
59 Layer layer = std::get<Layer>(std::move(current.node));
60 Box<X> resumed = std::visit(
61 overloaded{
62 [&handler]<operation Op>(impure_node<Op, X> &&node) -> Box<X> {
63 static_assert(
64 std::invocable<Handler &, Op>,
65 "run: handler does not cover every operation in "
66 "Sig -- see the impure_node<Op, X> named above in "
67 "the instantiation backtrace for which Op");
68 typename Op::response response =
69 std::invoke(handler, std::move(node.op));
70 return std::move(node.k)(std::move(response));
71 }},
72 std::move(layer.node));
73 current = *std::move(resumed);
74 }
75 return std::get<A>(std::move(current.node));
76}

78/** traceable_operation: an `operation` (freer.hpp) that additionally
79 * renders itself via the ordinary ADL-found `operator<<` idiom -- the same
80 * mechanism any other streamable type uses, not a bespoke customization
81 * point. Deliberately a separate concept, not an extra requirement folded
82 * into `operation` itself: `operation` is S03's landed contract and
83 * already has passing tests (`Get`/`Put` in freer.t.cpp) that do not
84 * define `operator<<`; broadening it here would break them.
85 */
86template <class Op>
87concept traceable_operation =
88 operation<Op> && requires(const Op &op, std::ostream &out) {
89 { out << op } -> std::same_as<std::ostream &>;
90 };
91
92/** trace: FD5's assertion vocabulary. One rendered string per visited
93 * operation, in visitation order. Tests assert "exactly these ops, in this
94 * order" as a plain `std::vector<std::string>` comparison -- terse, and
95 * reused verbatim by every later interpreter (S05-S08).
96 */
97using trace = std::vector<std::string>;
98
99/** render_operation(op) -> one trace entry: `op`'s own `operator<<`,
100 * captured as a `std::string` *before* the operation is moved into the
101 * handler -- never a pointer or reference surviving past the node it came
102 * from (the pitfall list's warning).
103 */
104template <traceable_operation Op>
105auto render_operation(const Op &op) -> std::string {
106 std::ostringstream out;
107 out << op;
108 return std::move(out).str();
109}

111/** tracing_handler<Handler> / tracing(handler, log): wrap `handler` (an
112 * `overloaded` per-op callable set, or any callable set covering `Sig`) so
113 * every operation passed through it is first rendered into `log` (FD5),
114 * then delegated to `handler` for the response. `log` is a caller-owned
115 * `trace&`; the adaptor itself is a thin pass-through, cheap to build
116 * fresh per `run_trace` call. This is FD5's "first handler, first mock":
117 * the reference interpreter for observational testing.
118 */
119template <class Handler>
120struct tracing_handler {
121 Handler handler;
122 trace *log;
123
124 template <traceable_operation Op>
125 constexpr auto operator()(Op &&op) ->
126 typename std::remove_cvref_t<Op>::response {
127 log->push_back(render_operation(op));
128 return std::invoke(handler, std::forward<Op>(op));
129 }
130};
131
132/** tracing(handler, log) -> tracing_handler<Handler>: see above. */
133template <class Handler>
134constexpr auto tracing(Handler &&handler, trace &log)
135 -> tracing_handler<std::remove_cvref_t<Handler>> {
136 return tracing_handler<std::remove_cvref_t<Handler>>{
137 std::forward<Handler>(handler), &log};
138}
139
140/** run_trace(handler, prog) -> {value, trace}: `run` through `tracing`'s
141 * reference handler (FD5) -- the vocabulary every later interpreter test
142 * (S05-S08) reuses to assert "exactly these ops, in this order" without
143 * inspecting the program's structure.
144 */
145template <class Sig, class A, class Handler>
146auto run_trace(Handler &&handler, Freer<Sig, A> &&prog) -> std::pair<A, trace> {
147 trace log;
148 A result = run<Sig, A>(tracing(std::forward<Handler>(handler), log),
149 std::move(prog));
150 return {std::move(result), std::move(log)};
151}
