14// std::pair<B, A> is the D11 dual of either.hpp's either<L, R>: pair is the
15// product (both components always present), either is the coproduct (exactly
16// one alternative is present). Slot convention matches throughout: the
17// non-value slot comes first, the value slot second — pair's env comonad has
18// extract = .second, either's monad has pure = make_right; zygo's
19// `pair<Helper, Result>` and either's `either<Stop, Continue>` follow the
20// same convention. Dual vocabulary:
21// introduction pair{a, b} <-> make_left / make_right
22// elimination .first / .second <-> match(e, on_left, on_right)
23// pairing fanout(f, g) <-> fanin(f, g)
24// one-sided map map_first(f, p) <-> map_left(f, e)
25// functor maps .second <-> maps Right
26//
27// Reference-assignment semantics differ between the two by construction, and
28// that is correct, not an inconsistency to fix: std::pair<T&, U> assigns
29// *through* the referent (both components always exist, so assignment can
30// always forward into them — this is standard std::pair behavior since
31// P2321R2/C++23, no extra work needed here), while either<L&, R> *rebinds*
32// (see either.hpp's Left<L&>/Right<R&>) because the active alternative can
33// change under assignment, so there may be no referent left to assign
34// through. Rebinding is the only coherent model for a sum; assigning through
35// is the only coherent model for a product.

39// -- Functor (maps .second, matching either's "maps Right") --

41template <class B, class A>
42struct PairFunctorImpl {
43 template <class Fn>
44 constexpr auto fmap(this auto &&, Fn &&fn, const std::pair<B, A> &p) {
45 using C = remove_cvref_t<std::invoke_result_t<Fn, const A &>>;
46 return std::pair<B, C>{p.first,
47 std::invoke(std::forward<Fn>(fn), p.second)};
48 }
49};
50
51template <class B, class A>
52struct PairFunctorMap : Functor<PairFunctorImpl<B, A>> {
53 using PairFunctorImpl<B, A>::fmap;
54};

56/** Functor instance for `std::pair<B, A>`: maps `.second`, leaving `.first`
57 * untouched — the dual of `either<L, R>`'s "maps Right".
58 */
59template <class B, class A>
60inline constexpr auto functor_typeclass<std::pair<B, A>> =
61 PairFunctorMap<B, A>{};

63// -- Comonad (the "env" / reader comonad: B is the fixed environment) --
64
65// `extract`/`duplicate`/`fmap` are generic over the value-slot type (their
66// own template parameter `X`), not fixed to the instance's own `A`; only the
67// environment `B` stays fixed (the env comonad never changes its
68// environment). This is needed so the Comonad CRTP's derived
69// `extend = fmap f . duplicate` can invoke `fmap` on the doubled structure
70// `std::pair<B, std::pair<B, X>>` that `duplicate` produces — mirroring how
71// OptionalMonadImpl's `bind` is generic in its own template parameter rather
72// than the instance-keying `VALUE_TYPE`. `A` only serves to key the
73// `comonad_typeclass<std::pair<B, A>>` lookup below.
74template <class B, class A>
75struct PairComonadImpl {
76 template <class X>
77 constexpr auto extract(this auto &&, const std::pair<B, X> &p)
78 -> const X & {
79 return p.second;
80 }
81
82 template <class X>
83 constexpr auto duplicate(this auto &&, const std::pair<B, X> &p)
84 -> std::pair<B, std::pair<B, X>> {
85 return std::pair<B, std::pair<B, X>>{p.first, p};
86 }
87
88 template <class Fn, class X>
89 constexpr auto fmap(this auto &&, Fn &&fn, const std::pair<B, X> &p) {
90 using C = remove_cvref_t<std::invoke_result_t<Fn, const X &>>;
91 return std::pair<B, C>{p.first,
92 std::invoke(std::forward<Fn>(fn), p.second)};
93 }
94};
95
96template <class B, class A>
97struct PairComonadMap : Comonad<PairComonadImpl<B, A>> {
98 using PairComonadImpl<B, A>::duplicate;
99 using PairComonadImpl<B, A>::extract;
100 using PairComonadImpl<B, A>::fmap;
101};

103/** The env/reader comonad instance for `std::pair<B, A>`: `extract` reads
104 * the value slot (`.second`), `duplicate` nests the whole pair back into the
105 * value slot alongside the unchanged environment (`.first`).
106 */
107template <class B, class A>
108inline constexpr auto comonad_typeclass<std::pair<B, A>> =
109 PairComonadMap<B, A>{};

113/** Maps `.first` only, leaving `.second` untouched; the dual of
114 * `map_left` (either.hpp).
115 */
116template <class Fn, class B, class A>
117constexpr auto map_first(Fn &&fn, const std::pair<B, A> &p) {
118 using C = remove_cvref_t<std::invoke_result_t<Fn, const B &>>;
119 return std::pair<C, A>{std::invoke(std::forward<Fn>(fn), p.first),
120 p.second};
121}

123/** The product introducer `<f, g>` (the pairing): returns a callable
124 * `x -> pair{f(x), g(x)}`; the dual of `fanin(f, g)` (either.hpp).
125 */
126template <class F, class G>
127constexpr auto fanout(F f, G g) {
128 return [f = std::move(f), g = std::move(g)](const auto &x) {
129 return std::pair{std::invoke(f, x), std::invoke(g, x)};
130 };
131}
