6// Compile-time type names without RTTI: type_name<T>() carves T's spelling
7// out of the enclosing function's signature macro, and short_type_name()
8// reduces that spelling to the leaf identifier a human wants in a diagram
9// label ("beman::execution::detail::then_t<...>" -> "then"). The full
10// spelling is compiler-specific (gcc, clang, and MSVC each format the
11// signature differently); the short form of a non-template library type is
12// stable across all three, which is what tests and labels should assert on.
13
14#include <string_view>

19/** The compiler's spelling of type @p T, at compile time. */
20template <typename T>
21constexpr auto type_name() -> std::string_view {
22#if defined(__clang__) || defined(__GNUC__)
23 // clang: "std::string_view smd::fixpoint::type_name() [T = int]"
24 // gcc: "constexpr std::string_view smd::fixpoint::type_name() [with T
25 // = int; std::string_view = std::basic_string_view<char>]"
26 constexpr std::string_view fn = __PRETTY_FUNCTION__;
27 constexpr auto start = fn.find("T = ") + 4;
28 constexpr auto semi = fn.find(';', start); // gcc's "; std::string_view..."
29 constexpr auto close = fn.rfind(']');
30 return fn.substr(start, (semi < close ? semi : close) - start);
31#elif defined(_MSC_VER)
32 // "class std::basic_string_view<...> __cdecl
33 // smd::fixpoint::type_name<int>(void)"
34 constexpr std::string_view fn = __FUNCSIG__;
35 constexpr auto start = fn.find("type_name<") + 10;
36 constexpr auto stop = fn.rfind(">(");
37 return fn.substr(start, stop - start);
38#else
39 return "unknown-type";
40#endif
41}

43/** Reduce a full type spelling to its leaf identifier: drop template
44 * arguments, then any namespace qualification, then a trailing "_t".
45 * The order matters — the namespace strip must not find a "::" inside
46 * template arguments, so those go first.
47 */
48constexpr auto short_type_name(std::string_view full) -> std::string_view {
49 if (auto angle = full.find('<'); angle != std::string_view::npos) {
50 full = full.substr(0, angle);
51 }
52 if (auto colons = full.rfind("::"); colons != std::string_view::npos) {
53 full = full.substr(colons + 2);
54 }
55 if (full.ends_with("_t")) {
56 full.remove_suffix(2);
57 }
58 return full;
59}

61/** The leaf-identifier spelling of type @p T. */
62template <typename T>
63constexpr auto short_type_name() -> std::string_view {
64 // The constexpr local is load-bearing, not style: it forces manifest
65 // constant evaluation. GCC 15.2's opportunistic fold of a plain
66 // `return short_type_name(type_name<T>());` miscompiles find('<') on
67 // the view into __PRETTY_FUNCTION__ -- it reports the character's
68 // offset in the underlying array (past the view's end), so the
69 // template-argument strip silently no-ops and "vector<int>" comes
70 // back instead of "vector". The constexpr evaluator gets it right;
71 // only the runtime fold is wrong. Callers wanting a short name for a
72 // type should come through here, not through the string overload on a
73 // type_name<T>() result, for the same reason.
74 constexpr std::string_view stripped = short_type_name(type_name<T>());
75 return stripped;
76}
