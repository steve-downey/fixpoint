18/** The trivial context: wraps exactly one value, with no additional effect.
19 * Functor/Applicative/Monad/Comonad all coincide with plain function
20 * application; Identity exists so schemes and law tests have a context with
21 * no effect to compare against ("the effect-free case"). Kept an aggregate
22 * (no user-declared constructors) so `Identity{x}` CTAD works, matching
23 * dist_cata's expected usage in later steps.
24 */
25template <class A>
26struct Identity {
27 using value_type = A;
28 A value;
29
30 friend constexpr bool operator==(const Identity &,
31 const Identity &) = default;
32};
33

36template <class A>
37struct IdentityFunctorImpl {
38 template <class Fn>
39 constexpr auto fmap(this auto &&, Fn &&fn, const Identity<A> &ident) {
40 using B = remove_cvref_t<std::invoke_result_t<Fn, const A &>>;
41 return Identity<B>{std::invoke(std::forward<Fn>(fn), ident.value)};
42 }
43};
44
45template <class A>
46struct IdentityFunctorMap : Functor<IdentityFunctorImpl<A>> {
47 using IdentityFunctorImpl<A>::fmap;
48};

50/** Functor instance for `Identity<A>`: maps the wrapped value. */
51template <class A>
52inline constexpr auto functor_typeclass<Identity<A>> = IdentityFunctorMap<A>{};

63 template <class FUNCTION_IN_CONTEXT, class ARGUMENT_IN_CONTEXT>

66 using Result = std::invoke_result_t<

72};

95 // Generic over the wrapped type (not fixed to the instance's own `A`),
96 // matching OptionalMonadImpl's `bind` — needed so the Monad CRTP's
97 // derived `join`/`apply`/`kleisli` can invoke `bind` on the doubled
98 // structure `Identity<Identity<X>>` they produce.
99 template <class X, class F>
100 constexpr auto bind(this auto &&, const Identity<X> &ma, F &&f) {
101 return std::invoke(std::forward<F>(f), ma.value);
102 }
103};

105template <class A>
106struct IdentityMonadMap : Monad<IdentityMonadImpl<A>> {
107 using IdentityMonadImpl<A>::bind;
108 using IdentityMonadImpl<A>::pure;
109};

111/** Monad instance for `Identity<A>`: bind is direct invocation on `.value`. */
112template <class A>
113inline constexpr auto monad_typeclass<Identity<A>> = IdentityMonadMap<A>{};

115// -- Comonad --
116
117// `extract`/`duplicate`/`fmap` are generic over the wrapped type (their own
118// template parameter `X`), not fixed to the instance's own `A` — needed so
119// the Comonad CRTP's derived `extend = fmap f . duplicate` can invoke `fmap`
120// on the doubled structure `Identity<Identity<X>>` that `duplicate`
121// produces. `A` only serves to key the `comonad_typeclass<Identity<A>>`
122// lookup below (mirroring how OptionalMonadImpl's `VALUE_TYPE` only keys
123// `monad_typeclass<optional<VALUE_TYPE>>` while `bind` is generic).
124template <class A>
125struct IdentityComonadImpl {
126 template <class X>
127 constexpr auto extract(this auto &&, const Identity<X> &wa) -> const X & {
128 return wa.value;
129 }
130
131 template <class X>
132 constexpr auto duplicate(this auto &&, const Identity<X> &wa)
133 -> Identity<Identity<X>> {
134 return Identity<Identity<X>>{wa};
135 }
136
137 template <class Fn, class X>
138 constexpr auto fmap(this auto &&, Fn &&fn, const Identity<X> &wa) {
139 using B = remove_cvref_t<std::invoke_result_t<Fn, const X &>>;
140 return Identity<B>{std::invoke(std::forward<Fn>(fn), wa.value)};
141 }
142};
143
144template <class A>
145struct IdentityComonadMap : Comonad<IdentityComonadImpl<A>> {
146 using IdentityComonadImpl<A>::duplicate;
147 using IdentityComonadImpl<A>::extract;
148 using IdentityComonadImpl<A>::fmap;
149};
150
151/** Comonad instance for `Identity<A>`: extract unwraps, duplicate re-wraps
152 * the whole context (there is nothing else to remember). */
153template <class A>
154inline constexpr auto comonad_typeclass<Identity<A>> = IdentityComonadMap<A>{};
