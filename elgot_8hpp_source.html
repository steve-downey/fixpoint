6// Elgot (co)algebras (design §7.8): a hylomorphism (refold) whose coalgebra
7// (elgot) or algebra (coelgot) gets an extra say over the recursion.
8//
9// elgot :: (f a -> a) -> (b -> Either a (f b)) -> b -> a
10// Equation: elgot phi psi s = case psi(s) of
11// Left(a) -> a
12// Right(layer) -> phi(fmapF(elgot phi psi,
13// layer))
14// The coalgebra can short-circuit *any* seed (including the very first one)
15// with a finished answer: D4's convention (Left = stop, Right = continue,
16// smd::typeclass::either, §5.2) applies here exactly as it does to apo
17// (apo.hpp) — Left = answer, stop; Right = one more layer to expand. Do NOT
18// swap this: the whole point of the scheme is that once psi(seed) returns
19// Left, psi is never called again for that seed or anything below it. This
20// is checked directly (not just by the final numeric answer, which a
21// plausible "expand one extra layer before discarding" bug can still get
22// right on many inputs — see elgot.t.cpp's invocation-counting test).
23//
24// coelgot :: ((a, f b) -> b) -> (a -> f a) -> a -> b
25// Equation: coelgot phi psi s = phi(s, fmapF(coelgot phi psi, psi(s)))
26// The algebra additionally receives the seed that produced the current
27// layer (the pair `(a, f b)` of the signature above, flattened into two
28// arguments per design §7.8's own C++ snippet: `algebra(const Seed&,
29// F<Result>)`). coelgot never short-circuits; psi(seed) is evaluated
30// exactly once per seed (bound to a local before fmap-ing it — a careless
31// transcription that calls psi twice, once to build the layer and again
32// inside the fmap, would silently double the coalgebra's side effects/cost
33// without changing the final answer for a pure psi).

42/** elgot :: (f a -> a) -> (b -> Either a (f b)) -> b -> a
43 *
44 * @tparam Result the fold's carrier (design D5: explicit)
45 * @param algebra F<Result> -> Result
46 * @param coalgebra Seed -> either<Result, F<Seed>> (D4: Left = answer, stop;
47 * Right = one more layer, continue)
48 */
49template <class Result, template <class> class F, class Algebra,
50 class Coalgebra, class Seed>
51constexpr auto elgot(const Algebra &algebra, const Coalgebra &coalgebra,
52 const Seed &seed) -> Result {
53 auto step = coalgebra(seed);
54 return smd::typeclass::match(
55 step, [](const Result &answer) -> Result { return answer; },
56 [&](const auto &layer) -> Result {
57 auto evaluated = layer_fmap(
58 [&](const Seed &child) -> Result {
59 return elgot<Result, F>(algebra, coalgebra, child);
60 },
61 layer);
62 return algebra(evaluated);
63 });
64}

68/** coelgot :: ((a, f b) -> b) -> (a -> f a) -> a -> b
69 *
70 * @tparam Result the fold's carrier (design D5: explicit)
71 * @param algebra (const Seed&, F<Result>) -> Result — the flattened pair;
72 * the seed that produced `layer` is passed alongside the
73 * already-folded children.
74 * @param coalgebra Seed -> F<Seed> (no short-circuit; always one more layer)
75 */
76template <class Result, template <class> class F, class Algebra,
77 class Coalgebra, class Seed>
78constexpr auto coelgot(const Algebra &algebra, const Coalgebra &coalgebra,
79 const Seed &seed) -> Result {
80 auto layer = coalgebra(seed); // evaluated exactly once per seed
81 auto evaluated = layer_fmap(
82 [&](const Seed &child) -> Result {
83 return coelgot<Result, F>(algebra, coalgebra, child);
84 },
85 layer);
86 return algebra(seed, evaluated);
87}
