20template <class FUNCTION, class... BOUND_ARGS>
21struct terminating_partial {
22 FUNCTION function;
23 std::tuple<BOUND_ARGS...> bound_args;
24
25 template <class NEXT_ARG>
26 auto operator()(NEXT_ARG &&next_arg) {
27 return invoke_or_extend(std::forward<NEXT_ARG>(next_arg),
28 std::index_sequence_for<BOUND_ARGS...>{});
29 }
30
31 template <class NEXT_ARG>
32 auto operator()(NEXT_ARG &&next_arg) const {
33 return invoke_or_extend_const(std::forward<NEXT_ARG>(next_arg),
34 std::index_sequence_for<BOUND_ARGS...>{});
35 }
36
37 private:
38 template <class NEXT_ARG, std::size_t... IDX>
39 auto invoke_or_extend(NEXT_ARG &&next_arg, std::index_sequence<IDX...>) {
40 if constexpr (std::invocable<FUNCTION &, BOUND_ARGS &..., NEXT_ARG>) {
41 return std::invoke(function, std::get<IDX>(bound_args)...,
42 std::forward<NEXT_ARG>(next_arg));
43 } else {
44 using NEXT_PARTIAL = terminating_partial<FUNCTION, BOUND_ARGS...,
45 remove_cvref_t<NEXT_ARG>>;
46 return NEXT_PARTIAL{
47 function,
48 std::tuple_cat(std::move(bound_args),
49 std::tuple<remove_cvref_t<NEXT_ARG>>{
50 std::forward<NEXT_ARG>(next_arg)})};
51 }
52 }
53
54 template <class NEXT_ARG, std::size_t... IDX>
55 auto invoke_or_extend_const(NEXT_ARG &&next_arg,
56 std::index_sequence<IDX...>) const {
57 if constexpr (std::invocable<const FUNCTION &, const BOUND_ARGS &...,
58 NEXT_ARG>) {
59 return std::invoke(function, std::get<IDX>(bound_args)...,
60 std::forward<NEXT_ARG>(next_arg));
61 } else {
62 using NEXT_PARTIAL = terminating_partial<FUNCTION, BOUND_ARGS...,
63 remove_cvref_t<NEXT_ARG>>;
64 return NEXT_PARTIAL{
65 function,
66 std::tuple_cat(bound_args,
67 std::tuple<remove_cvref_t<NEXT_ARG>>{
68 std::forward<NEXT_ARG>(next_arg)})};
69 }
70 }
71};

73template <class FUNCTION>
74auto make_terminating_partial(FUNCTION &&function) {
75 using STORED_FUNCTION = remove_cvref_t<FUNCTION>;
76 return terminating_partial<STORED_FUNCTION>{
77 std::forward<FUNCTION>(function), std::tuple<>{}};
78}

80// apply_chain: left-folds `self.ap` over the remaining effectful arguments.
81// A free function (not a class member) so it is unaffected by the
82// class-hierarchy access-control subtlety noted below: with deducing-`this`,
83// a call of the form `self.member(...)` is checked as an access through
84// `self`'s *reified* type, not merely by where the call is lexically
85// written. Once `self`'s type is reached by crossing a `protected`
86// inheritance step two or more class-template layers below the class
87// declaring `member`, a private *or* protected class member becomes
88// inaccessible even though the call site is textually inside the
89// declaring class -- exactly the shape of the `Applicative` structural
90// chain here (Functor <- Applicative <- Monad, each `protected Impl`).
91// Keeping this helper as a free function sidesteps the issue entirely: it
92// only ever calls the already-public `self.ap`.
93template <class SELF, class ACCUMULATED>
94auto apply_chain(SELF &&, ACCUMULATED &&accumulated) {
95 return std::forward<ACCUMULATED>(accumulated);
96}
97
98template <class SELF, class ACCUMULATED, class NEXT_ARGUMENT,
99 class... REST_ARGUMENTS>
100auto apply_chain(SELF &&self, ACCUMULATED &&accumulated,
101 NEXT_ARGUMENT &&next_argument,
102 REST_ARGUMENTS &&...rest_arguments) {
103 auto next = self.ap(std::forward<ACCUMULATED>(accumulated),
104 std::forward<NEXT_ARGUMENT>(next_argument));
105 if constexpr (sizeof...(REST_ARGUMENTS) == 0) {
106 return next;
107 } else {
108 return apply_chain(std::forward<SELF>(self), std::move(next),
109 std::forward<REST_ARGUMENTS>(rest_arguments)...);
110 }
111}

115// See the typeclass contract in detail/typeclass_base.hpp.
116// Applicative-specific: the primitives are pure and apply. invoke is
117// derived from them via terminating partial application, but an Impl
118// may override invoke for different semantics or efficiency. Other
119// ops (lift/ap/map/zip_with/discard_*) are derived. Structurally
120// chains onto Functor (via applicative_functor_adapter) to expose
121// fmap/replace.
122
123/** Adapter: supplies a single-argument `fmap` so an Applicative can also
124 * serve as a Functor base, defined as `self.invoke(fn, x)` so it defers to
125 * the most-derived object's `invoke` (respecting any Impl-provided override).
126 */
127template <class Impl>
128struct applicative_functor_adapter : protected Impl {
129 using Impl::apply;
130 using Impl::pure;
131
132 template <class FUNCTION, class ARGUMENT>
133 auto fmap(this auto &&self, FUNCTION &&function, ARGUMENT &&argument) {
134 return self.invoke(std::forward<FUNCTION>(function),
135 std::forward<ARGUMENT>(argument));
136 }
137};

139/** CRTP base for Applicative instances.
140 * `Impl` must provide `pure(value)` and `apply(f_in_context, arg_in_context)`.
141 * All other operations are derived. Structurally chains onto `Functor` (via
142 * `applicative_functor_adapter`) so the full Functor surface (`fmap`,
143 * `replace`) is also available on an Applicative object.
144 */
145template <class Impl>
146struct Applicative : protected Functor<applicative_functor_adapter<Impl>> {
147 static_assert(
148 !std::is_same_v<Impl, std::false_type>,
149 "No applicative_typeclass<T> specialization found. "
150 "Specialize smd::typeclass::applicative_typeclass<T> for "
151 "your type T and provide pure(...) and apply(...) operations.");
152 // Alternate-core: pure + apply are the primitives; invoke and all others
153 // derive from them. fmap/replace come from the Functor base.
154 using Functor<applicative_functor_adapter<Impl>>::fmap;
155 using Functor<applicative_functor_adapter<Impl>>::replace;
156 using Impl::apply;
157 using Impl::pure;

163 FIRST_ARGUMENT &&first_argument,
164 REST_ARGUMENTS &&...rest_arguments) {

170 /** Lifts `function` into the applicative and applies it to one or more
171 * effectful arguments left-to-right, producing a single effectful result.
172 * @param function A plain callable; it is wrapped with `pure` internally.
173 * @param first_argument First effectful argument (e.g., optional or
174 * vector).
175 * @param rest_arguments Additional effectful arguments, if any.
176 */
177 template <class FUNCTION, class FIRST_ARGUMENT, class... REST_ARGUMENTS>
178 auto invoke(this auto &&self, FUNCTION &&function,
179 FIRST_ARGUMENT &&first_argument,
180 REST_ARGUMENTS &&...rest_arguments) {
181 using SELF = std::remove_reference_t<decltype(self)>;
182 using IMPL_BASE =
183 std::conditional_t<std::is_const_v<SELF>, const Impl, Impl>;
184
185 if constexpr (requires(IMPL_BASE &impl) {
186 impl.invoke(
187 std::forward<FUNCTION>(function),
188 std::forward<FIRST_ARGUMENT>(first_argument),
189 std::forward<REST_ARGUMENTS>(rest_arguments)...);
190 }) {
191 return static_cast<IMPL_BASE &>(self).invoke(
192 std::forward<FUNCTION>(function),
193 std::forward<FIRST_ARGUMENT>(first_argument),
194 std::forward<REST_ARGUMENTS>(rest_arguments)...);
195 } else {
196 auto lifted_function = self.pure(detail::make_terminating_partial(
197 std::forward<FUNCTION>(function)));
198 return detail::apply_chain(
199 self,
200 self.ap(std::move(lifted_function),
201 std::forward<FIRST_ARGUMENT>(first_argument)),
202 std::forward<REST_ARGUMENTS>(rest_arguments)...);
203 }
204 }

319template <class VALUE_TYPE>
320struct OptionalApplicativeImpl {
321 template <class VALUE>
322 auto pure(this auto &&, VALUE &&value)
323 -> std::optional<remove_cvref_t<VALUE>> {
324 return std::optional<remove_cvref_t<VALUE>>{std::forward<VALUE>(value)};
325 }
326
327 template <class FUNCTION_IN_CONTEXT, class ARGUMENT_IN_CONTEXT>
328 auto apply(this auto &&, FUNCTION_IN_CONTEXT &&function,
329 ARGUMENT_IN_CONTEXT &&argument) {
330 using Result =
331 std::invoke_result_t<decltype(*function), decltype(*argument)>;
332
333 if (!function || !argument) {
334 return std::optional<remove_cvref_t<Result>>{};
335 }
336
337 return std::optional<remove_cvref_t<Result>>{
338 std::invoke(*std::forward<FUNCTION_IN_CONTEXT>(function),
339 *std::forward<ARGUMENT_IN_CONTEXT>(argument))};
340 }
341};

343template <class VALUE_TYPE>
344struct OptionalApplicativeMap
345 : Applicative<OptionalApplicativeImpl<VALUE_TYPE>> {
346 using OptionalApplicativeImpl<VALUE_TYPE>::apply;
347 using OptionalApplicativeImpl<VALUE_TYPE>::pure;
348};
