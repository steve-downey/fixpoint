6// hoist, prepro, postpro (design §7.4): natural-transformation plumbing.
7//
8// A *natural transformation* here is any polymorphic function object
9// callable as `e(F<X>) -> F<X>` (or `-> G<X>` for hoist) for *every* X — not
10// just the one concrete X a particular call site happens to instantiate.
11// Fixture transformations therefore need a templated call operator, e.g.:
12//
13// struct identity_nat {
14// template <class A>
15// constexpr auto operator()(const NatF<A> &layer) const -> NatF<A> {
16// return layer;
17// }
18// };
19//
20// (design §4's `cap_layer` example is the same shape). A lambda with a
21// concrete layer parameter type is *not* a natural transformation: prepro
22// below applies `e` at `F<Fix<F>>` (via hoist, over whole subtrees), not at
23// the carrier type the surrounding algebra/coalgebra uses, so a
24// non-templated call operator will fail to compile or silently pick the
25// wrong overload.
26

36/** hoist :: (forall x. f x -> g x) -> Fix f -> Fix g
37 *
38 * Equation: hoist e = fold_fix(wrap_fix<G> ∘ e)
39 *
40 * Retags every layer of the tree from functor @p F to functor @p G via the
41 * natural transformation @p e, applied bottom-up (a fold whose algebra is
42 * "transform this layer, then wrap it"). For the common endo case (@p G ==
43 * @p F, i.e. rewriting a tree's layers without changing its functor), the
44 * target functor still has to be given explicitly since it cannot be
45 * deduced from @p e alone: call as `hoist<F>(e, t)`.
46 *
47 * @tparam G target functor (D5: explicit, cannot be deduced from @p e)
48 * @tparam F source functor (deduced from @p tree)
49 * @param e F<X> -> G<X> for every X
50 */
51template <template <class> class G, template <class> class F, class Nat>
52constexpr auto hoist(const Nat &e, const Fix<F> &tree) -> Fix<G> {
53 return fold_fix<Fix<G>>(
54 [&](const F<Fix<G>> &layer) -> Fix<G> { return wrap_fix<G>(e(layer)); },
55 tree);
56}

60/** prepro (Fokkinga) :: (forall x. f x -> f x) -> (f a -> a) -> Fix f -> a
61 *
62 * Equation: prepro e φ = φ ∘ fmapF (prepro e φ ∘ hoist<F>(e)) ∘ unfix
63 *
64 * A fold whose algebra sees, at each child, the result of first
65 * rewriting the *entire child subtree* via `hoist<F>(e)` and then
66 * recursing prepro on the rewritten subtree. This is the "transform on
67 * the way down" dual to postpro's "transform on the way out".
68 *
69 * Cumulative-cost note: because each recursive step re-hoists the whole
70 * remaining subtree before descending into it, a node at depth k has had
71 * @p e applied to it k times by the time prepro's algebra sees its
72 * layer (once per ancestor, each hoist itself walking every layer below
73 * it). This is the classical (and expensive) reading of prepro — Fokkinga's
74 * equation is transcribed literally here, not fused/memoized. At the small
75 * tree sizes exercised by this module's tests it is not observable, but it
76 * is worth knowing before reaching for prepro over anything but small or
77 * shallow trees.
78 *
79 * @tparam Result carrier type (D5: explicit)
80 * @tparam F functor (deduced from @p tree)
81 * @param e F<X> -> F<X> for every X (a natural transformation, endo case)
82 * @param algebra F<Result> -> Result
83 */
84template <class Result, template <class> class F, class Nat, class Algebra>
85constexpr auto prepro(const Nat &e, const Algebra &algebra, const Fix<F> &tree)
86 -> Result {
87 const auto &layer = unwrap_fix(tree);
88 auto evaluated = layer_fmap(
89 [&](const Fix<F> &child) -> Result {
90 return prepro<Result>(e, algebra, hoist<F>(e, child));
91 },
92 layer);
93 return algebra(evaluated);
94}

98/** postpro :: (forall x. f x -> f x) -> (a -> f a) -> a -> Fix f
99 *
100 * Equation: postpro e ψ = fix ∘ fmapF (hoist<F>(e) ∘ postpro e ψ) ∘ ψ
101 *
102 * Dual to prepro: an unfold whose coalgebra's children are first unfolded
103 * as usual, then the *entire resulting subtree* is rewritten via
104 * `hoist<F>(e)` before being grafted in. Same cumulative-cost shape as
105 * prepro, mirrored: a node at depth k has been hoisted k times by the time
106 * it is grafted into the final tree.
107 *
108 * @tparam F functor (D5: explicit, cannot be deduced from @p seed alone)
109 * @param e F<X> -> F<X> for every X (a natural transformation, endo case)
110 * @param coalgebra Seed -> F<Seed>
111 */
112template <template <class> class F, class Nat, class Coalgebra, class Seed>
113constexpr auto postpro(const Nat &e, const Coalgebra &coalgebra,
114 const Seed &seed) -> Fix<F> {
115 auto layer = coalgebra(seed);
116 auto expanded = layer_fmap(
117 [&](const Seed &child) -> Fix<F> {
118 return hoist<F>(e, postpro<F>(e, coalgebra, child));
119 },
120 layer);
121 return wrap_fix<F>(std::move(expanded));
122}
