6// dyna, codyna, chrono (design §7.6): the three course-of-values *refold*
7// fusions. Each is histo/futu fused with refold (recursion_schemes.hpp) so
8// that no intermediate Fix<F> tree is ever materialized — unlike
9// "histo(unfold_fix(psi, seed))" or "fold_fix(futu(psi, seed))", which would
10// build the whole Fix<F> tree first and then walk it a second time, these
11// three interleave the coalgebra and algebra in a single refold pass, one
12// F-layer at a time.
13//
14// dyna = histo . ana, fused via refold
15// codyna = cata . futu, fused via refold
16// chrono = histo . futu, fused via refold
17//
18// codyna/chrono share a coalgebra shape: refold needs a single-step
19// Seed -> F<Seed> coalgebra, but futu's own coalgebra is Seed -> F<Free<F,
20// Seed>> (it may emit several F-layers per step, as a Free chunk). `unroll`
21// bridges the two: given a Free<F, Seed> chunk, it peels exactly one
22// F-layer — resuming the original coalgebra at a Pure leaf, or handing back
23// an already-built Roll layer unchanged — and refold itself supplies the
24// recursion over that layer's children (each still a Free<F, Seed>).
25// `unroll` never fully resolves a chunk to Fix<F> itself (unlike futu's own
26// futu_worker, futu.hpp) — it only ever peels one layer per call, exactly
27// what refold's coalgebra parameter needs.

39/** Implementation detail of codyna/chrono: peels one F-layer off a
40 * Free<F, Seed> chunk (as futu's coalgebra would emit). A Pure leaf resumes
41 * the original coalgebra at its seed; a Roll layer is handed back verbatim
42 * (by value — it holds further Free<F,Seed> children for refold to keep
43 * unrolling on its own next call).
44 */
45template <template <class> class F, class Coalgebra, class Seed>
46constexpr auto unroll(const Coalgebra &coalgebra, const Free<F, Seed> &chunk)
47 -> F<Free<F, Seed>> {
48 return std::visit(
49 overloaded{
50 [&](const Seed &s) -> F<Free<F, Seed>> { return coalgebra(s); },
51 [&](const F<Free<F, Seed>> &layer) -> F<Free<F, Seed>> {
52 return layer;
53 },
54 },
55 chunk.node);
56}

59/** dyna :: (f (Cofree f b) -> b) -> (a -> f a) -> a -> b
60 * histo . ana, fused: never builds the intermediate Fix<F>.
61 * @param algebra F<Cofree<F, Result>> -> Result
62 * @param coalgebra Seed -> F<Seed>
63 */
64template <class Result, template <class> class F, class Algebra,
65 class Coalgebra, class Seed>
66constexpr auto dyna(const Algebra &algebra, const Coalgebra &coalgebra,
67 const Seed &seed) -> Result {
68 using Carrier = Cofree<F, Result>;
69 auto combined = [&](const F<Carrier> &layer) -> Carrier {
70 return Carrier{algebra(layer), layer};
71 };
72 return extract(refold<Carrier, F>(combined, coalgebra, seed));
73}

77/** codyna :: (f b -> b) -> (a -> f (Free f a)) -> a -> b
78 * cata . futu, fused: never builds the intermediate Fix<F>.
79 * @param algebra F<Result> -> Result
80 * @param coalgebra Seed -> F<Free<F, Seed>>

82template <class Result, template <class> class F, class Algebra,
83 class Coalgebra, class Seed>
84constexpr auto codyna(const Algebra &algebra, const Coalgebra &coalgebra,
85 const Seed &seed) -> Result {
86 auto unroll_step =
87 [&coalgebra](const Free<F, Seed> &chunk) -> F<Free<F, Seed>> {
