15// Traversable pattern invariants:
16// - Instances are single lookup objects that provide traverse(F, T).
17// - transpose is a derived object operation implemented from
18// traverse(identity).
19// - Dispatch happens through a provided object or
20// traversable_typeclass<Concrete>.
21// - Traversal must preserve container shape while transposing structure and
22// context.
23
24/** CRTP base for Traversable instances.
25 * `Impl` must provide `traverse(applicative, f, container)` and declare
26 * `element_type`. All other operations (`transpose`, `for_each`,
27 * `traverse_with`, `transpose_with`) are derived.
28 */
29template <class Impl>
30struct Traversable : protected Impl {
31 static_assert(!std::is_same_v<Impl, std::false_type>,
32 "No traversable_typeclass<T> specialization found. "
33 "Specialize smd::typeclass::traversable_typeclass<T> for "
34 "your type T, "
35 "provide traverse(applicative, F, T), and declare 'using "
36 "element_type = T;'.");
37 static_assert(
38 requires { typename Impl::element_type; },
39 "Traversable Impl must declare 'using element_type = T;' "
40 "so that transpose() and traverse_with() can deduce the element type.");
41 // Alternate-core: Impl::traverse is the primitive; transpose is derived
42 // from it. A transpose-primitive Impl would shadow transpose instead.
43 using Impl::traverse;
44 using element_type = typename Impl::element_type;
45
46 /** Applies `function` to each element and transposes the resulting
47 * effects; the applicative is inferred from the return type of `function`.
48 */
49 template <class T, class F>
50 auto for_each(this auto &&self, T &&value, F &&function) {
51 using Context =
52 remove_cvref_t<std::invoke_result_t<F, const element_type &>>;
53 const auto &applicative = applicative_typeclass<Context>;
54 return self.traverse(applicative, std::forward<F>(function),
55 std::forward<T>(value));
56 }
57
58 /** Transposes a structure of effectful values into one outer effect
59 * containing the structure. The element type must itself be an
60 * applicative context.
61 */
62 template <class T>
63 auto transpose(this auto &&self, T &&value) {
64 using Context = element_type;
65 const auto &applicative = applicative_typeclass<Context>;
66 return self.traverse(
67 applicative, [](auto &&x) { return std::forward<decltype(x)>(x); },
68 std::forward<T>(value));
69 }
70
71 /** Traverses using a different traversable instance; applicative is
72 * inferred from the return type of `function`.
73 */
74 template <class TRAVERSABLE_MAP, class T, class F>
75 auto traverse_with(this auto &&, const TRAVERSABLE_MAP &traversable_map,
76 F &&function, T &&value) {
77 using Context = remove_cvref_t<std::invoke_result_t<
78 F, const typename remove_cvref_t<TRAVERSABLE_MAP>::element_type &>>;
79 const auto &applicative = applicative_typeclass<Context>;
80 return traversable_map.traverse(applicative, std::forward<F>(function),
81 std::forward<T>(value));
82 }
83
84 /** Traverses using explicit traversable and applicative instances. */
85 template <class TRAVERSABLE_MAP, class APPLICATIVE_MAP, class T, class F>
86 auto traverse_with(this auto &&, const TRAVERSABLE_MAP &traversable_map,
87 const APPLICATIVE_MAP &applicative_map, F &&function,
88 T &&value) {
89 return traversable_map.traverse(
90 applicative_map, std::forward<F>(function), std::forward<T>(value));
91 }
92
93 /** Transposes using a different traversable instance; applicative is
94 * inferred from the container's element type.
95 */
96 template <class TRAVERSABLE_MAP, class T>
97 auto transpose_with(this auto &&self,
98 const TRAVERSABLE_MAP &traversable_map, T &&value) {
99 return self.traverse_with(
100 traversable_map,
101 [](auto &&x) { return std::forward<decltype(x)>(x); },
102 std::forward<T>(value));
103 }
104};

106/** Typeclass lookup variable for Traversable; specialize for each container
107 * type. */
108template <class T>
109inline constexpr auto traversable_typeclass = std::false_type{};
110
111/** @brief Maps `function` over `value`, traverses effects left-to-right,
112 * and preserves container shape.
113 *
114 * @param function A callable returning an applicative effect for each element.
115 * @param value The traversable container to process.
116 * @return The container shape transposed into the applicative effect.
117 */
118template <class F, class T>
119auto traverse(F &&function, T &&value) {
120 const auto &map = traversable_typeclass<remove_cvref_t<T>>;
121 using element_type = typename remove_cvref_t<decltype(map)>::element_type;
122 using Context =
123 remove_cvref_t<std::invoke_result_t<F, const element_type &>>;
124 const auto &applicative = applicative_typeclass<Context>;
125 return map.traverse(applicative, std::forward<F>(function),
126 std::forward<T>(value));
127}
