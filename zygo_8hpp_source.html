6// Zygomorphism (design §7.3): a fold with an auxiliary "helper" fold running
7// alongside it. The main algebra sees, at every child position, both the
8// helper's value and the main fold's result for that child.
9//
10// Implemented via pairing (the banana-split construction): a single
11// fold_fix with carrier std::pair<Helper, Result> and combined algebra
12// `λx. pair(helper(fmapF(first, x)), main(x))`, then `.second` of the
13// overall result.

24/** zygo :: (f b -> b) -> (f (b, a) -> a) -> t -> a
25 *
26 * Equation: zygo f g = second ∘ fold_fix(λx. pair(f(fmapF(first, x)), g(x)))
27 *
28 * Convention (design §7.3, matches dist_zygo in S12): the carrier pair is
29 * `{helper value, main value}` — **helper first, main second**. This is the
30 * opposite slot order from para's `std::pair<Fix<F>, Result>` (original
31 * subtree first, fold result second); the two pairs serve different
32 * purposes and are not to be confused.
33 *
34 * @tparam Result main fold's carrier (D5: explicit)
35 * @tparam Helper helper fold's carrier (D5: explicit)
36 * @param helper F<Helper> -> Helper
37 * @param main F<std::pair<Helper, Result>> -> Result
38 */
39template <class Result, class Helper, template <class> class F, class HelperAlg,
40 class MainAlg>
41constexpr auto zygo(const HelperAlg &helper, const MainAlg &main,
42 const Fix<F> &tree) -> Result {
43 using Carrier = std::pair<Helper, Result>;
44 auto combined = [&](const F<Carrier> &layer) -> Carrier {
45 // A *second* fmap over the already fold_fix-mapped layer, projecting
46 // just the helper component at each child position — inherent to
47 // the pairing construction, not a fusion opportunity (design §7.3
48 // pitfall).
49 auto helper_layer = layer_fmap(
50 [](const Carrier &c) -> Helper { return c.first; }, layer);
51 return Carrier{helper(helper_layer), main(layer)};
52 };
53 return fold_fix<Carrier>(combined, tree).second;
54}
