6// The Free/Cofree pairing (design doc FD8): a scripted mock as a Cofree
7// value over the CO-signature -- the dual of the request functor
8// signature<Ops...> generates (freer.hpp) -- paired against a Freer program
9// by a second, synchronous interpreter alongside run/run_trace
10// (freer_run.hpp). Where signature<Ops...>::type<X> is a VARIANT of
11// impure_node<Op,X> (exactly one pending request), cosignature<Sig>::type<X>
12// is a PRODUCT of one responder per operation (an answer ready for every
13// possible request) -- function-space shaped, per FD8's strict-Cofree
14// correction: Cofree's tail is by-value (cofree.hpp:37), so the only finite
15// value that represents "an answer to every possible next request" is a
16// callable per operation, not an enumerated case.
17//
18// The co-signature's Functor instance is LAZY -- the exact dual of FD6's
19// Coyoneda instance (freer.hpp): fmap post-composes onto each stored
20// responder WITHOUT invoking it. This laziness is what makes unfold_cofree
21// (unfold_cofree.hpp) terminate over the co-signature after building exactly
22// one node, regardless of the coalgebra -- the coalgebra's recursive calls
23// are only forced when a responder is actually invoked, at pairing time.
24// Same registration route as FD11/D-A (a constrained functor_typeclass
25// partial specialization, disjoint from freer.hpp's coyoneda_layer marker),
26// same FD4 capture-ownership rule applied to its instance.

47/** responder<Op, X>: one entry of a co-signature product (FD8) -- a stored
48 * callable answering Op with (Op::response, X), where X is the co-signature
49 * functor's own recursive position (the mock's next state).
50 *
51 * std::function, not one_shot (D-B's move-only continuation type): the mock
52 * is conceptually re-askable (FD8's own text) -- Cofree's tail is a
53 * by-value/shared structure, unlike Freer's move-only one-shot continuation
54 * -- so the simpler, copyable, multi-shot std::function shape is both
55 * sufficient and the more natural fit. Recorded choice, not a deviation:
56 * FD8 leaves the responder representation open ("plain std::function-shaped
57 * ... OR match D-B's one-shot type ... pick the simpler spelling").
58 */
59template <operation Op, class X>
60struct responder {
61 std::function<std::pair<typename Op::response, X>(Op)> respond;
62};

66// ===========================================================================
67// cosignature_layer<Pack>: the product-of-responders struct, built once from
68// a flattened operation pack -- reused by both cosignature<signature<Ops...>>
69// and cosignature<row<Sigs...>> below. Mirrors freer_row.hpp's own
70// "flattened pack -> concrete type" step (ops_pack_variant_t there; the
71// struct-of-responders here instead of a variant is the only thing that
72// differs, per that file's own forward note to this step).
73// ===========================================================================
74
75template <class Pack>
76struct cosignature_layer;
77
78template <operation... Ops>
79struct cosignature_layer<ops_pack<Ops...>> {
80 /** The co-signature layer functor: one responder<Op,X> base per
81 * operation in Pack. Marker typedefs (dual of freer.hpp's
82 * coyoneda_signature/coyoneda_value_type, deliberately
83 * co-signature-specific names so the two constrained
84 * functor_typeclass partial specializations never overlap) key the
85 * generic Functor instance below.
86 */
87 template <class X>
88 struct type : responder<Ops, X>... {
89 using co_signature_ops = ops_pack<Ops...>;
90 using co_signature_value_type = X;
91 };
92
93 /** Lazy fmap over the whole product (FD8, the dual of FD6's Coyoneda
94 * fmap, freer.hpp): build ONE NEW responder per operation, each
95 * post-composing `fn` onto the OLD stored responder WITHOUT invoking
96 * it -- the new responder's own callable, when eventually invoked
97 * (by a pairing interpreter, never here), calls the old responder and
98 * applies `fn` to the X it returns. Each operation's responder is an
99 * independent base subobject of `layer`, so extracting/moving one
100 * does not disturb the others -- no aliasing between the pack
101 * expansion's N calls below.
102 */
103 template <class Fn, class X>
104 static constexpr auto fmap_layer(Fn fn, type<X> &&layer)
105 -> type<std::remove_cvref_t<std::invoke_result_t<Fn, X &&>>> {
106 using Y = std::remove_cvref_t<std::invoke_result_t<Fn, X &&>>;
107 return type<Y>{fmap_one<Ops>(
108 fn, std::move(static_cast<responder<Ops, X> &>(layer)))...};
109 }
110
111 private:
112 // Capture-ownership rule (FD4): `fn` is captured by value/copy (never
113 // by reference) into the new responder's closure -- that closure is
114 // STORED past this call's return (the entire point of the co-signature
115 // being lazy), so a reference-captured `fn` would dangle the moment
116 // fmap_layer returns, exactly the hazard FD4 documents for the
117 // Coyoneda side. `fn` is duplicated once per operation (a genuinely
118 // different closure owns each copy), unlike CoyonedaFunctorImpl::fmap's
119 // single std::visit-selected alternative -- the co-signature is a
120 // PRODUCT (every responder coexists), not a variant, so `fn` must be
121 // copyable here.
122 template <operation Op, class Fn, class X>
123 static constexpr auto fmap_one(Fn fn, responder<Op, X> &&r)
124 -> responder<Op, std::remove_cvref_t<std::invoke_result_t<Fn, X &&>>> {
125 using Y = std::remove_cvref_t<std::invoke_result_t<Fn, X &&>>;
126 return responder<Op, Y>{
127 [old = std::move(r.respond),
128 fn](Op op) mutable -> std::pair<typename Op::response, Y> {
129 auto [response, next] = old(std::move(op));
130 return {std::move(response), std::invoke(fn, std::move(next))};
131 }};
132 }
133};

135} // namespace detail

137/** cosignature<Sig>: FD8's co-signature functor for a closed signature or
138 * row Sig -- the dual of signature<Ops...>/row<Sigs...> (freer.hpp,
139 * freer_row.hpp). `Sig` may be a bare `signature<Ops...>` or a
140 * `row<Sigs...>`; the row case flattens over every member signature's
141 * operations (mirroring row's own FLATTENED node_variant encoding, not a
142 * product-of-products), reusing freer_row.hpp's detail::row_ops_pack_t
143 * verbatim to recover the flattened Ops pack.
144 *
145 * Primary template intentionally left undefined: naming cosignature<T> for
146 * a T that is neither shape fails here with an incomplete-type diagnostic
147 * rather than silently doing the wrong thing (mirrors freer_row.hpp's
148 * signature_ops_pack).
149 */
150template <class Sig>
151struct cosignature;
152
153template <operation... Ops>
154struct cosignature<signature<Ops...>>
155 : detail::cosignature_layer<detail::ops_pack<Ops...>> {};
156
157template <class... Sigs>
158struct cosignature<row<Sigs...>>
159 : detail::cosignature_layer<detail::row_ops_pack_t<Sigs...>> {};

161/** co_signature_layer<T>: T is (an instantiation of) some
162 * cosignature<Sig>::type<X> -- the dual of freer.hpp's coyoneda_layer,
163 * keying the generic lazy Functor instance below via the SAME registration
164 * route (FD11/D-A: a constrained functor_typeclass partial specialization).
165 * Deliberately co-signature-specific marker names, never overlapping
166 * coyoneda_layer's (a Layer cannot satisfy both concepts at once).
167 */
168template <class T>
169concept co_signature_layer = requires {
170 typename T::co_signature_ops;
171 typename T::co_signature_value_type;
172};

178// -- The generic co-signature Functor instance (FD8, dual of FD6) --
179//
180// One instance object serves every co-signature layer, exactly as
181// CoyonedaFunctorMap (freer.hpp) serves every signature layer: `fmap`
182// recovers the flattened operation pack from the layer's own marker
183// typedef and delegates to detail::cosignature_layer<Pack>::fmap_layer,
184// which never invokes a stored responder -- see that function's own
185// comment for the laziness/capture-ownership argument. Consuming-only, no
186// const-lvalue overload: the responders are stored by value inside the
187// product, but fmap must still route a `Layer&&` through so the moved-from
188// old responders can be relocated into the new closures without copying an
189// entire product of std::function objects.
190template <class Layer>
191 requires smd::fixpoint::co_signature_layer<Layer>
192struct CoSignatureFunctorImpl {
193 template <class Fn>
194 constexpr auto fmap(this auto &&, Fn &&fn, Layer &&layer)
195 -> smd::fixpoint::detail::
196 cosignature_layer<typename Layer::co_signature_ops>::template type<
197 remove_cvref_t<std::invoke_result_t<
198 Fn, typename Layer::co_signature_value_type &&>>> {
199 using Pack = typename Layer::co_signature_ops;
200 return smd::fixpoint::detail::cosignature_layer<Pack>::fmap_layer(
201 std::forward<Fn>(fn), std::move(layer));
202 }
203};
204
205template <class Layer>
206 requires smd::fixpoint::co_signature_layer<Layer>
207struct CoSignatureFunctorMap : Functor<CoSignatureFunctorImpl<Layer>> {
208 using CoSignatureFunctorImpl<Layer>::fmap;
209};

232/** pair_run(mock, prog) -> (A, B): the Free/Cofree pairing (FD8, Piponi
233 * "Cofree Meets Free") -- a second, synchronous interpreter alongside
234 * run/run_trace (freer_run.hpp), driven by a scripted Mock instead of a
235 * handler function.
236 *
237 * A LOOP, not recursion (mirrors run's own shape/rationale exactly,
238 * freer_run.hpp): each iteration handles one suspended operation, selects
239 * the matching responder from the CURRENT mock state's tail (a direct
240 * upcast to responder<Op, Mock<Sig,B>> -- one base subobject per
241 * operation, so selection is O(1), no scanning), resumes the program's
242 * continuation with the response, and REPLACES the mock state with the
243 * responder's own next state -- so the C++ call stack never grows with the
244 * number of effects a program performs, same as run.
245 *
246 * Sig, A, B must all be given explicitly at every call site (the
247 * deduction-limitation discovery is universal across this whole layer --
248 * S04/S05 handoffs): Sig cannot be deduced from a Freer<Sig,A> parameter,
249 * and B cannot be deduced from a Mock<Sig,B> (=Cofree<...,B>) parameter for
250 * the same alias-transparency reason.
251 */
252template <class Sig, class A, class B>
253auto pair_run(Mock<Sig, B> &&mock, Freer<Sig, A> &&prog) -> std::pair<A, B> {
254 using X = Freer<Sig, A>;
255 using Layer = typename Sig::template type<X>;
256 using M = Mock<Sig, B>;
257
258 Freer<Sig, A> current = std::move(prog);
259 M state = std::move(mock);
260 while (!is_pure(current)) {
261 Layer layer = std::get<Layer>(std::move(current.node));
262 Box<X> resumed = std::visit(
263 overloaded{
264 [&state]<operation Op>(impure_node<Op, X> &&node) -> Box<X> {
265 auto &r = static_cast<responder<Op, M> &>(state.tail);
266 auto [response, next_state] = r.respond(std::move(node.op));
267 state = std::move(next_state);
268 return std::move(node.k)(std::move(response));
269 }},
270 std::move(layer.node));
271 current = *std::move(resumed);
272 }
273 return {std::get<A>(std::move(current.node)), std::move(state.head)};
274}

276/** pair_run_trace(mock, prog) -> (A, B, trace): pair_run through the SAME
277 * `trace` type run_trace uses (freer_run.hpp) -- FD8's "tests read
278 * identically against either interpreter" claim made literal: an op needs
279 * only its own operator<< (traceable_operation, freer_run.hpp) to be
280 * assertable here, exactly as for run_trace. A short, self-contained loop
281 * rather than a thin wrapper over pair_run (contrast run_trace, built by
282 * wrapping run's handler argument via `tracing`): pair_run has no handler
283 * argument to wrap -- the mock's responders play that role -- so recording
284 * would otherwise have to rebuild an equivalent "tracing Cofree" adaptor
285 * for no benefit over a second small loop.
286 */
287template <class Sig, class A, class B>
288auto pair_run_trace(Mock<Sig, B> &&mock, Freer<Sig, A> &&prog)
289 -> std::tuple<A, B, trace> {
290 using X = Freer<Sig, A>;
291 using Layer = typename Sig::template type<X>;
292 using M = Mock<Sig, B>;
293
294 trace log;
295 Freer<Sig, A> current = std::move(prog);
296 M state = std::move(mock);
297 while (!is_pure(current)) {
298 Layer layer = std::get<Layer>(std::move(current.node));
299 Box<X> resumed = std::visit(
300 overloaded{[&state, &log]<traceable_operation Op>(
301 impure_node<Op, X> &&node) -> Box<X> {
302 log.push_back(render_operation(node.op));
303 auto &r = static_cast<responder<Op, M> &>(state.tail);
304 auto [response, next_state] = r.respond(std::move(node.op));
305 state = std::move(next_state);
306 return std::move(node.k)(std::move(response));
307 }},
308 std::move(layer.node));
309 current = *std::move(resumed);
310 }
311 return {std::get<A>(std::move(current.node)), std::move(state.head),
312 std::move(log)};
313}
