6// overloaded<Ts...> — aggregate visitor for std::visit.
7//
8// Usage:
9// std::visit(overloaded{
10// [](int x) { ... },
11// [](std::string s) { ... },
12// }, v);
13//
14// The consteval catch-all fires a static_assert at compile time if std::visit
15// encounters an alternative not covered by the explicit cases. This turns
16// variant exhaustiveness into a hard compile error rather than a silent
17// default/no-op. Adding a new alternative to a variant without handling it
18// everywhere is caught immediately.
19//
20// No explicit deduction guide is needed: C++20 CTAD for aggregates deduces
21// overloaded<F1, F2, ...> from the constructor arguments.
22
23namespace smd::fixpoint {
24
25template <typename... Ts>
26struct overloaded : Ts... {
27 using Ts::operator()...;
28
29 consteval void operator()(auto) const {
30 static_assert(false,
31 "overloaded: unhandled variant alternative — add a case");
32 }
33};

35} // namespace smd::fixpoint
