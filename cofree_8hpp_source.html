6// Cofree comonad (design §5.3): annotates every node of an F-shaped tree
7// with a value of type A at the head, keeping one F-layer of already
8// annotated children as the tail. Cofree<F, Result> is histo's fold carrier
9// (§7.5, histo.hpp): each child position an algebra sees carries the
10// *entire* annotated history of that subtree, not just its final result.
11//
12// Complete-type reasoning mirrors Fix<F> (design §4): F's recursive
13// positions are Box<Cofree<F,A>> (e.g. Succ<Cofree<F,A>>{Box<Cofree<F,A>>
14// pred}), which is what makes `F<Cofree<F,A>> tail` a complete member type
15// even though Cofree<F,A> is still being defined.

30/** Cofree<F, A>: an F-tree where every node is annotated with an A.
31 * @tparam F unary template functor (the base functor being annotated)
32 * @tparam A the annotation/history type at each node
33 */
34template <template <class> class F, class A>
35struct Cofree {
36 A head; // the annotation at this node
37 F<Cofree<F, A>> tail; // one functor layer of annotated children
38
39 // Hand-written (not = default): this is the exact type the clang-22
40 // "incomplete type" cascade (dist_laws.t.cpp/gprepro.t.cpp) traces
41 // through — forming F<WWR> for WWR = Cofree<F,X> (generalized.hpp's
42 // gcata_worker_t/gprepro_worker_t) requires std::variant to ask
43 // is_trivially_destructible_v<Succ<WWR>>, which (per Clang, more
44 // eagerly than GCC) forces this defaulted friend's deleted-ness check,
45 // which requires Cofree<F,X> itself complete — a cycle. A plain friend
46 // body sidesteps this: it is only instantiated when actually called.
47 friend constexpr auto operator==(const Cofree &lhs, const Cofree &rhs)
48 -> bool {
49 return lhs.head == rhs.head && lhs.tail == rhs.tail;
50 }
51};
52
53/** extract(c) -> const A& : the annotation at this node. */
54template <template <class> class F, class A>
55constexpr auto extract(const Cofree<F, A> &c) -> const A & {
56 return c.head;
57}
58
59/** unwrap_cofree(c) -> const F<Cofree<F,A>>& : one layer of annotated
60 * children.
61 */
62template <template <class> class F, class A>
63constexpr auto unwrap_cofree(const Cofree<F, A> &c) -> const F<Cofree<F, A>> & {
64 return c.tail;
65}

74template <template <class> class F, class A>
75struct CofreeFunctorImpl {
76 // Explicit (not deduced) trailing return type: fmap recurses on itself
77 // through the nested lambda below, and a deduced `auto` return type
78 // cannot be used before its own deduction completes, even when the
79 // recursive call is one level down inside a lambda (GCC 16 diagnoses
80 // this as "use of ... before deduction of 'auto'"). Naming the return
81 // type up front breaks the cycle.
82 template <class Fn>
83 constexpr auto fmap(this auto &&self, Fn &&fn,
84 const smd::fixpoint::Cofree<F, A> &c)
85 -> smd::fixpoint::Cofree<
86 F, remove_cvref_t<std::invoke_result_t<Fn, const A &>>> {
87 using B = remove_cvref_t<std::invoke_result_t<Fn, const A &>>;
88 auto mapped_tail = smd::fixpoint::layer_fmap(
89 [&self, &fn](const smd::fixpoint::Cofree<F, A> &child)
90 -> smd::fixpoint::Cofree<F, B> { return self.fmap(fn, child); },
91 c.tail);
92 return smd::fixpoint::Cofree<F, B>{std::invoke(fn, c.head),
93 std::move(mapped_tail)};
94 }
95};
96
97template <template <class> class F, class A>
98struct CofreeFunctorMap : Functor<CofreeFunctorImpl<F, A>> {
99 using CofreeFunctorImpl<F, A>::fmap;
100};

102/** Functor instance for Cofree<F, A>: maps every head throughout the whole

106template <template <class> class F, class A>
107inline constexpr auto functor_typeclass<smd::fixpoint::Cofree<F, A>> =

110// -- Comonad --
111
112// extract/duplicate/fmap are generic over their own element-type parameter
113// `X`, not fixed to the instance's own `A` (S03 handoff's discovery,
114// design §6.3): the derived `extend = fmap . duplicate` needs `fmap` to
115// consume the *doubled* structure `Cofree<F, Cofree<F, X>>` that
116// `duplicate` produces, which is not `Cofree<F, A>` for the class-level
117// `A`. Only the outer functor `F` stays fixed at the class level; `A` here
118// serves solely to key the `comonad_typeclass<Cofree<F, A>>` specialization
119// below.
120template <template <class> class F, class A>
121struct CofreeComonadImpl {
122 template <class X>
123 constexpr auto extract(this auto &&, const smd::fixpoint::Cofree<F, X> &c)
124 -> const X & {
125 return c.head;
126 }
127
128 template <class X>
129 constexpr auto duplicate(this auto &&self,
130 const smd::fixpoint::Cofree<F, X> &c)
131 -> smd::fixpoint::Cofree<F, smd::fixpoint::Cofree<F, X>> {
132 auto mapped_tail = smd::fixpoint::layer_fmap(
133 [&self](const smd::fixpoint::Cofree<F, X> &child) {
134 return self.duplicate(child);
135 },
136 c.tail);
137 return smd::fixpoint::Cofree<F, smd::fixpoint::Cofree<F, X>>{
138 c, std::move(mapped_tail)};
139 }
140
141 // Explicit trailing return type for the same reason as
142 // CofreeFunctorImpl::fmap above: it is directly self-recursive (through
143 // the nested lambda), and a deduced `auto` return type cannot be used
144 // before its own deduction completes.
145 template <class Fn, class X>
146 constexpr auto fmap(this auto &&self, Fn &&fn,
147 const smd::fixpoint::Cofree<F, X> &c)
148 -> smd::fixpoint::Cofree<
149 F, remove_cvref_t<std::invoke_result_t<Fn, const X &>>> {
150 using B = remove_cvref_t<std::invoke_result_t<Fn, const X &>>;
151 auto mapped_tail = smd::fixpoint::layer_fmap(
152 [&self, &fn](const smd::fixpoint::Cofree<F, X> &child)
153 -> smd::fixpoint::Cofree<F, B> { return self.fmap(fn, child); },
154 c.tail);
155 return smd::fixpoint::Cofree<F, B>{std::invoke(fn, c.head),
156 std::move(mapped_tail)};
157 }
158};

160template <template <class> class F, class A>
161struct CofreeComonadMap : Comonad<CofreeComonadImpl<F, A>> {
162 using CofreeComonadImpl<F, A>::duplicate;
163 using CofreeComonadImpl<F, A>::extract;
164 using CofreeComonadImpl<F, A>::fmap;
165};
