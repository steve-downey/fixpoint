18/** Verify the Applicative identity law: `pure(id) <*> v == v`. */
19template <class CONTEXT>
20auto check_applicative_identity_law(const CONTEXT &value) -> bool {
21 const auto &applicative = applicative_typeclass<remove_cvref_t<CONTEXT>>;
22 auto result = applicative.ap(
23 applicative.pure([](const auto &x) { return x; }), value);
24 return result == value;
25}
26
27/** Verify the Applicative homomorphism law: `pure(f) <*> pure(x) == pure(f x)`.
28 */
29template <class CONTEXT, class FUNCTION, class VALUE>
30auto check_applicative_homomorphism_law(const FUNCTION &function,
31 const VALUE &value) -> bool {
32 const auto &applicative = applicative_typeclass<remove_cvref_t<CONTEXT>>;
33 auto left =
34 applicative.ap(applicative.pure(function), applicative.pure(value));
35 auto right = applicative.pure(std::invoke(function, value));
36 return left == right;
37}

54/** Foldable implementation for Sequence<V>: fold_map walks values
55 * left-to-right. */
56template <class VALUE_TYPE>
57struct TestSequenceFoldableImpl {
58 template <class FUNCTION>
59 auto fold_map(this auto &&, FUNCTION &&function,
60 const test::Sequence<VALUE_TYPE> &sequence) {
61 using Result =
62 remove_cvref_t<std::invoke_result_t<FUNCTION, const VALUE_TYPE &>>;
63 return std::ranges::fold_left(
64 sequence.values, monoid_identity<Result>(),
65 [&](Result acc, const VALUE_TYPE &value) {
66 return monoid_combine(std::move(acc),
67 std::invoke(function, value));
68 });
69 }
70};

82/** Traversable implementation for the public Identity<V> (smd/typeclass/
83 * identity.hpp). Test-only: there is no general Traversable typeclass
84 * instance for Identity in the library proper, only here, since traversing
85 * an effect-free context is a law-testing convenience, not a scheme need. */
86template <class VALUE_TYPE>
87struct TestIdentityTraversableImpl {
88 using element_type = VALUE_TYPE;
89
90 template <class APPLICATIVE, class FUNCTION>
91 auto traverse(this auto &&, const APPLICATIVE &applicative,
92 FUNCTION &&function, const Identity<VALUE_TYPE> &identity) {
93 return applicative.invoke(
94 [](auto &&value) {
95 using U = remove_cvref_t<decltype(value)>;
96 return Identity<U>{std::forward<decltype(value)>(value)};
97 },
98 std::invoke(std::forward<FUNCTION>(function), identity.value));
99 }
100};
