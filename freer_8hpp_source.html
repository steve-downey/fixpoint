6// The closed-world Coyoneda signature layer (design doc FD1-FD3, FD6, FD11,
7// FD12): Freer<Sig, A> is Free<F,A> (free.hpp) instantiated at the layer
8// functor a closed effect signature generates, plus the generic Functor
9// instance that makes `bind`/`fmap` on that layer O(1) closure
10// post-composition instead of a per-signature hand-written traversal.
11//
12// No new fixpoint: Freer is an alias over the existing Free, and (FD3's
13// windfall, confirmed at instantiation by this header's own tests) the
14// existing functor_typeclass<Free<F,A>>/monad_typeclass<Free<F,A>>
15// specializations already select for Freer<Sig,A> with zero new
16// registration -- only the *layer*-level Functor instance
17// (functor_typeclass<Sig::type<X>>) is new, registered per FD11's decision
18// (D-A, option (a)): a constrained variable-template partial specialization
19// keyed on a Coyoneda-specific marker (signature<Ops...>::type<X>'s nested
20// `coyoneda_signature`/`coyoneda_value_type` typedefs), so the existing
21// implicit-mode `layer_fmap` dispatch (fmap.hpp) resolves to it without any
22// call-site change.

42/** unit: the response of operations that return no information.
43 * A bespoke empty type rather than std::monostate: monostate carries
44 * "empty variant alternative" connotations and comparison semantics we do
45 * not want conflated with "trivial response".
46 */
47struct unit {
48 friend constexpr auto operator==(const unit &, const unit &)
49 -> bool = default;
50};

52/** An effect operation: a movable request payload that names the type a
53 * handler must supply to resume the program. The `response` member type is
54 * the entire protocol -- the per-constructor index of the Haskell GADT
55 * FD1 dissolves into a closed variant, spelled the only way C++ can spell
56 * it.
57 */
58template <class Op>
59concept operation = std::movable<Op> && requires { typename Op::response; };
60
61/** impure_node<Op, X>: one suspended request. Holds the operation payload
62 * and the one-shot continuation from the handler's response to the rest of
63 * the program.
64 *
65 * The continuation returns Box<X>, not X, for the same complete-type
66 * reasoning as Fix/Free/Cofree: at the point this member is declared,
67 * X = Free<..., A> is still incomplete, and Box is the completeness
68 * firewall. `one_shot` is itself a second completeness firewall of the
69 * same shape (one_shot.hpp) -- FD12's bespoke continuation representation,
70 * uniform on every toolchain (D-B: libc++ at LLVM 23 does not declare
71 * std::move_only_function at all).
72 *
73 * one_shot's rvalue-qualified call operator makes one-shot-ness a type
74 * property: the continuation can only be invoked on an rvalue, i.e.
75 * std::move(k)(r). See FD4.
76 */
77template <operation Op, class X>
78struct impure_node {
79 Op op;
80 one_shot<Box<X>(typename Op::response)> k;
81};
82
83/** signature<Ops...>: a closed effect signature. Its nested member template
84 * `type` is the layer functor handed to Free -- the closed-world Coyoneda
85 * of the signature.
86 *
87 * A member template rather than an alias template: alias templates as
88 * template-template arguments sit on P0522 matching, where GCC and Clang
89 * still diverge in practice; the nested class template is the portable
90 * spelling (FD3).
91 *
92 * `coyoneda_signature`/`coyoneda_value_type` are the nested typedefs FD11's
93 * decision (D-A) requires: they let the generic Functor instance recover,
94 * from the layer type alone, the owning signature and the recursive
95 * position it needs to build the mapped layer's result type.
96 * `coyoneda_signature` is deliberately signature-specific (not a generic
97 * name like "recursive_position") so the constrained specialization it
98 * keys never overlaps the Free<F,A> specialization or any pure-data
99 * functor instance in the library.
100 */
101template <operation... Ops>
102struct signature {
103 template <class X>
104 struct type {
105 using coyoneda_signature = signature<Ops...>;
106 using coyoneda_value_type = X;
107 using node_variant = std::variant<impure_node<Ops, X>...>;
108
109 node_variant node;
110 };
111};

113/** Freer<Sig, A>: the freer monad over a closed signature -- the existing
114 * Free applied to the signature's Coyoneda layer. No new fixpoint.
115 */
116template <class Sig, class A>
117using Freer = Free<Sig::template type, A>;
118
119/** Concept satisfied by a Coyoneda layer type produced by some
120 * `signature<Ops...>::type<X>` -- keys the generic Functor instance's
121 * constrained partial specialization (FD11, D-A). Deliberately keyed on
122 * the Coyoneda-specific `coyoneda_signature`/`coyoneda_value_type`
123 * typedefs rather than a generic name, per FD11's binding constraint.
124 */
125template <class T>
126concept coyoneda_layer = requires {
127 typename T::coyoneda_signature;
128 typename T::coyoneda_value_type;
129 typename T::node_variant;
130};
131
132/** send<Sig>(op): lift one operation into Freer<Sig, Op::response> -- the
133 * unit of the adjunction (FD7). The continuation is the identity-into-Pure
134 * -- Coyoneda's `id` -- so the program suspends with the request and
135 * resumes with the handler's response as its value.
136 *
137 * The continuation is a `one_shot<Box<X>(R)>` (FD12/D-B), not a bare
138 * lambda or `std::move_only_function`: `impure_node`'s `k` member is
139 * declared at that type, and a plain lambda converts to it implicitly via
140 * `one_shot`'s templated constructor.
141 *
142 * Variant construction: unlike S01's/S03's defensive precedent (which
143 * never actually attempted this), FD7's literal aggregate-init spelling
144 * over `Sig::template type<X>` -- relying on `node_variant`'s converting
145 * constructor to pick the alternative matching the exact `impure_node<Op,
146 * X>` prvalue -- compiles cleanly here on both gcc-16 and clang-23,
147 * verified by a standalone probe before landing this; no `in_place_type`
148 * needed for this particular construction (see the S04 handoff for the
149 * probe and a hypothesis on why this call site differs from S03's
150 * `CoyonedaFunctorImpl::fmap`, which does still need it).
151 */
152template <class Sig, operation Op>
153auto send(Op op) -> Freer<Sig, typename Op::response> {
154 using R = typename Op::response;
155 using X = Freer<Sig, R>;
156 return roll_free<Sig::template type>(
157 typename Sig::template type<X>{impure_node<Op, X>{
158 std::move(op), one_shot<Box<X>(R)>([](R r) -> Box<X> {
159 return make_box<X>(pure_free<Sig::template type>(std::move(r)));
160 })}});
161}

167// -- The generic Coyoneda Functor instance (FD6, granularity correction) --
168//
169// One instance object serves every closed-world signature: `fmap` takes
170// the whole layer `Sig::type<X>&&`, std::visits the variant, and per
171// alternative post-composes `fn` onto the stored one_shot continuation --
172// O(1), no traversal of the rest of the program. Consuming-only: no
173// const-lvalue overload exists, or ever should (FD4) -- the impure layer's
174// continuation is move-only, so a const path could not compile; leaving it
175// unwritten turns "no const path" into a clear error at the call site
176// instead of template vomit.
177
178template <class Layer>
179 requires smd::fixpoint::coyoneda_layer<Layer>
180struct CoyonedaFunctorImpl {
181 template <class Fn>
182 constexpr auto fmap(this auto &&, Fn &&fn, Layer &&layer) ->
183 typename Layer::coyoneda_signature::template type<remove_cvref_t<
184 std::invoke_result_t<Fn, typename Layer::coyoneda_value_type &&>>> {
185 using X = typename Layer::coyoneda_value_type;
186 using Y = remove_cvref_t<std::invoke_result_t<Fn, X &&>>;
187 using Sig = typename Layer::coyoneda_signature;
188 using Result = typename Sig::template type<Y>;
189 using ResultVariant = typename Result::node_variant;
190
191 // Capture-ownership rule (FD4): the closure handed to std::visit
192 // owns `fn` by move -- never `[&fn]` -- because the *inner* per-node
193 // continuation it builds is stored past this call's return. The
194 // outer visitor lambda itself runs synchronously, exactly once
195 // (std::visit dispatches to exactly one live alternative), so
196 // moving `fn` into it here and moving it again into the inner
197 // closure below is safe: there is only ever one live alternative
198 // to move from, no double-move race (contrast free.hpp's own
199 // consuming bind/fmap, which branch instead of using
200 // std::visit(overloaded{...}) for a different reason -- there, one
201 // *externally shared* fn is captured by two eagerly-constructed
202 // visitor alternatives; here each alternative's inner continuation
203 // captures its *own* per-node `k`, so the hazard those comments
204 // describe does not apply to this visit).
205 return std::visit(
206 smd::fixpoint::overloaded{
207 [fn = std::forward<Fn>(fn)]<smd::fixpoint::operation Op>(
208 smd::fixpoint::impure_node<Op, X> &&node) mutable
209 -> Result {
210 return Result{ResultVariant{
211 std::in_place_type<smd::fixpoint::impure_node<Op, Y>>,
212 smd::fixpoint::impure_node<Op, Y>{
213 std::move(node.op),
214 smd::fixpoint::one_shot<smd::fixpoint::Box<Y>(
215 typename Op::response)>(
216 [k = std::move(node.k), fn = std::move(fn)](
217 typename Op::response r) mutable
218 -> smd::fixpoint::Box<Y> {
219 return smd::fixpoint::make_box<Y>(
220 std::invoke(
221 fn, *std::move(k)(std::move(r))));
222 })}}};
223 }},
224 std::move(layer.node));
225 }
226};

228template <class Layer>
229 requires smd::fixpoint::coyoneda_layer<Layer>
230struct CoyonedaFunctorMap : Functor<CoyonedaFunctorImpl<Layer>> {
231 using CoyonedaFunctorImpl<Layer>::fmap;
232};

234/** Functor instance for every closed-world Coyoneda layer: user signatures
235 * never write fmap (FD6). Registered per FD11 (D-A, option (a)): a
236 * constrained variable-template partial specialization of
237 * `functor_typeclass`, so the existing implicit-mode `layer_fmap`
238 * (fmap.hpp) that free.hpp's consuming Free fmap/bind already call finds
239 * it with no call-site change. Functor only: the Monad is Free over the
240 * layer (the FD3 windfall), so no layer-level `monad_typeclass`
241 * specialization is written here.
242 */
243template <class Layer>
244 requires smd::fixpoint::coyoneda_layer<Layer>
245inline constexpr auto functor_typeclass<Layer> = CoyonedaFunctorMap<Layer>{};
