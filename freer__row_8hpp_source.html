6// Rows: variant-of-signatures composition, recovering openness after FD1
7// closed each individual signature (design doc FD1's closing paragraph,
8// section 6 forward reference, FD10's Clock+Network example).
9//
10// row<Sigs...> is a composed signature: its layer `type<X>` exposes the
11// SAME nested-typedef surface signature<Ops...>::type<X> does
12// (coyoneda_signature/coyoneda_value_type/node_variant/node), but
13// node_variant is FLATTENED over every member signature's operations --
14// one std::variant<impure_node<Op,X>...>, not a variant of per-signature
15// variants. That is what makes the S03 generic Coyoneda Functor instance
16// (functor_typeclass<coyoneda_layer>, freer.hpp) cover rows for free: no
17// new fmap is written here, or ever needs to be (FD6/FD11).
18//
19// member<Op, SigOrRow> is the C++ image of the 2015 paper's `Member`
20// typeclass -- concept-checked injection, not the paper's open-union
21// Typeable/index machinery (FD1 explicitly rejects that). discharge<Sig>
22// is the handler-as-adaptor: it walks a row program consuming it, handles
23// the operations belonging to one member signature inline via a
24// synchronous handler (the same shape run's handler is), and re-emits
25// every other operation via send into the smaller row (Row minus Sig).
26

142/** row<Sigs...>: a composed signature over several independent member
143 * signatures. Its layer `type<X>` exposes exactly the nested-typedef
144 * surface signature<Ops...>::type<X> does (coyoneda_signature,
145 * coyoneda_value_type, node_variant, and a `node` data member of that
146 * variant type) -- so the S03 generic Coyoneda Functor instance covers
147 * rows with zero new fmap, and `run`/`run_trace`/`send` (freer_run.hpp,
148 * freer.hpp) all work over a row unmodified (see the S05 handoff for
149 * confirmation this actually holds).
150 *
151 * `coyoneda_signature` names `row<Sigs...>` itself (not any one member
152 * signature) -- the mapped-result layer S03's generic fmap builds is
153 * `row<Sigs...>::template type<Y>`, per that instance's own contract; this
154 * is why `row` needs its own `template<class X> struct type`, exactly like
155 * `signature` (FD6/FD11's whole point, verified to fall out here).
156 */
157template <class... Sigs>
158struct row {
159 static_assert(
160 detail::pack_all_distinct_v<detail::row_ops_pack_t<Sigs...>>,
161 "row<Sigs...>: two (or more) of the row's member signatures share "
162 "an operation type -- every operation type must be unique across "
163 "all member signatures of a row (a shared operation type would "
164 "collide as a duplicate std::variant alternative and make "
165 "injection ambiguous)");
166
167 /** The row's own flattened operation pack. Doubles as the marker the
168 * `row_type` concept keys on (send<Row>'s overload, below) and as what
169 * `member_ops_pack` (the `member` concept) reads for a row argument.
170 */
171 using row_member_signatures = detail::row_ops_pack_t<Sigs...>;
172
173 template <class X>
174 struct type {
175 using coyoneda_signature = row<Sigs...>;
176 using coyoneda_value_type = X;
177 using node_variant =
178 detail::ops_pack_variant_t<detail::row_ops_pack_t<Sigs...>, X>;
179
180 node_variant node;
181 };
182};

184/** row_type<T>: T is (an instantiation of) `row<Sigs...>`. Keys the
185 * send<Row> overload below -- deliberately NOT the same marker `member`
186 * checks membership against, so that overload stays viable (and wins
187 * partial ordering against freer.hpp's unconstrained send<Sig,Op>) even
188 * when Op turns out not to be a member; see that overload's comment.

190template <class T>
191concept row_type = requires { typename T::row_member_signatures; };

226/** member<Op, SigOrRow>: Op is an operation of the signature or row
227 * `SigOrRow` -- the C++ image of the 2015 paper's `Member` typeclass
228 * (concept-checked injection, not open-union index machinery, FD1).
229 * `SigOrRow` may be a bare `signature<Ops...>` or a `row<Sigs...>`.
230 */
231template <class Op, class SigOrRow>
232concept member = operation<Op> &&
233 detail::op_in_pack_v<Op, detail::member_ops_pack_t<SigOrRow>>;

235/** send<Row>(op): send<Sig,Op> (freer.hpp) extended to any row containing
236 * Op as an operation of one of its member signatures.
237 *
238 * Deliberately constrained on `row_type<Row>` alone, NOT on
239 * `member<Op,Row>`: freer.hpp's send<Sig,Op> has no constraint tying Op to
240 * Sig at all (it is viable for any Sig/Op pair, valid or not), so if THIS
241 * overload were instead constrained on `member<Op,Row>`, an invalid Op
242 * would SFINAE this overload away and leave freer.hpp's unconstrained one
243 * as the only candidate -- which would then instantiate and fail deep
244 * inside node_variant's converting constructor, exactly the "variant
245 * vomit" the design doc warns against. Constraining on `row_type<Row>`
246 * instead keeps this overload viable (and, by constraint subsumption in
247 * C++20 partial ordering, strictly more specialized than freer.hpp's
248 * unconstrained template whenever Row actually is a row) regardless of
249 * whether Op is a member -- so THIS overload, not freer.hpp's, is always
250 * the one selected for a row argument, and the readable diagnostic below
251 * is what actually fires for a non-member Op. Verified empirically (not
252 * just reasoned through) before landing; see the S05 handoff for the
253 * verbatim diagnostic and confirmation this overload -- not freer.hpp's --
254 * is the one selected for valid calls too.
255 */
256template <row_type Row, operation Op>
257auto send(Op op) -> Freer<Row, typename Op::response> {
258 static_assert(member<Op, Row>,
259 "send<Row>(op): Op is not an operation of any member "
260 "signature of Row");
261 using R = typename Op::response;
262 using X = Freer<Row, R>;
263 return roll_free<Row::template type>(
264 typename Row::template type<X>{impure_node<Op, X>{
265 std::move(op), one_shot<Box<X>(R)>([](R r) -> Box<X> {
266 return make_box<X>(pure_free<Row::template type>(std::move(r)));
267 })}});
268}

317/** discharge<Sig>(handler, prog): reinterpret a Freer<Row,A> program,
318 * handling every operation belonging to the member signature `Sig`
319 * inline via `handler` (a complete, synchronous handler for `Sig` alone --
320 * the same callable-set shape `run`'s handler is), and re-emitting every
321 * other operation via `send` into the smaller row (Row minus Sig).
322 *
323 * Consuming and self-recursive through STORED continuations: every
324 * closure built here that outlives this call -- the re-emitted
325 * continuation wrapper in the non-matching branch -- owns its captures
326 * (`k`, `handler`) BY MOVE, never by reference (FD4's capture-ownership
327 * rule). Built directly on `std::visit(overloaded{...}, ...)` over the
328 * row's node_variant (run's pattern, freer_run.hpp), not on
329 * `tracing_handler`: discharge conditionally re-emits rather than always
330 * delegating-and-recording (S04 handoff's forward note).
331 *
332 * Sig, Row, and A must all be given explicitly at every call site: Sig
333 * does not appear in any function parameter at all, and Row/A are hidden
334 * inside the `Freer<Row,A>` alias, which template argument deduction
335 * cannot invert (S04's discovery -- confirmed to apply identically here).
336 * Handler deduces normally from the function argument, like `run`.
337 */
338template <class Sig, class Row, class A, class Handler>
339auto discharge(Handler &&handler, Freer<Row, A> &&prog)
340 -> Freer<detail::row_without_t<Sig, Row>, A> {
341 using SmallRow = detail::row_without_t<Sig, Row>;
342 using X = Freer<Row, A>;
343 using SmallX = Freer<SmallRow, A>;
344 using Layer = typename Row::template type<X>;
345
346 Freer<Row, A> current = std::move(prog);
347 if (is_pure(current)) {
348 return pure_free<SmallRow::template type>(
349 std::get<A>(std::move(current.node)));
350 }
351
352 Layer layer = std::get<Layer>(std::move(current.node));
353 return std::visit(
354 overloaded{[handler = std::forward<Handler>(handler)]<operation Op>(
355 impure_node<Op, X> &&node) mutable -> SmallX {
356 if constexpr (member<Op, Sig>) {
357 // Sig's own op: handle inline, then keep discharging
358 // whatever the resumed continuation yields next.
359 typename Op::response response =
360 std::invoke(handler, std::move(node.op));
361 X next = *std::move(node.k)(std::move(response));
362 return discharge<Sig, Row, A>(std::move(handler),
363 std::move(next));
364 } else {
365 static_assert(
366 member<Op, SmallRow>,
367 "discharge<Sig>: Op belongs to neither Sig nor the "
368 "remaining row -- every operation of Row must be a "
369 "member of exactly one of its signatures");
370 // Not Sig's op: re-emit into the smaller row. Built
371 // DIRECTLY -- the same literal aggregate-init shape
372 // send<Row> uses (freer.hpp's send, this file's
373 // send<Row>) -- rather than via
374 // smd::typeclass::mbind(send<SmallRow>(op), ...):
375 // mbind/FreeMonadImpl::bind's generic recursive
376 // traversal (free.hpp) re-threads its continuation
377 // through a nested, NON-mutable closure (`[self, fn =
378 // std::forward<Fn>(fn)](auto &&child) { return
379 // self.bind(child, fn); }`) that the *template*
380 // instantiates for both the Pure and Roll cases of
381 // whatever `child` might structurally be -- including
382 // the Roll case's OWN re-capture of `fn`, which, once
383 // `fn` has been forced const by that outer non-mutable
384 // closure, requires COPY-constructing `fn`. A
385 // continuation that owns a move-only `one_shot` (as
386 // this one must, holding `node.k`) has no copy
387 // constructor, so that generic path cannot be
388 // instantiated with it -- confirmed empirically (see
389 // the S05 handoff for the diagnostic) before landing
390 // this direct construction instead. Building the node
391 // by hand sidesteps `bind`'s multi-child-fanout
392 // generality entirely, which a single suspended
393 // Coyoneda node never needed in the first place (FD6:
394 // layer_fmap for the impure layer is O(1) closure
395 // post-composition, not a traversal).
396 //
397 // This closure is STORED inside the returned SmallRow
398 // node's continuation -- it outlives this call's stack
399 // frame entirely, so both captures are owned by move
400 // (FD4), never by reference.
401 using R = typename Op::response;
402 return roll_free<SmallRow::template type>(
403 typename SmallRow::template type<SmallX>{
404 impure_node<Op, SmallX>{
405 std::move(node.op),
406 one_shot<Box<SmallX>(R)>([k = std::move(node.k),
407 handler = std::move(
408 handler)](R r) mutable
409 -> Box<SmallX> {
410 X next = *std::move(k)(std::move(r));
411 return make_box<SmallX>(discharge<Sig, Row, A>(
412 std::move(handler), std::move(next)));
413 })}});
414 }
415 }},
416 std::move(layer.node));
417}
